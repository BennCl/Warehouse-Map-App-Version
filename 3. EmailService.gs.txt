const SPREADSHEET_ID_EMAIL = '1wLTDovDcceXdVEK4H0R-2L4OkrAgW8f61pKYYWIPCA4'; 

var EmailService = {

  // --- 1. CORE SENDING FUNCTION ---
  _send: function(to, subject, htmlBody, ccEmail) {
    if (!to) return;
    const options = { htmlBody: htmlBody, name: "Wickes Holiday Portal" };
    if (ccEmail) options.cc = ccEmail; // Explicitly set CC
    try {
      MailApp.sendEmail(to, subject, "", options);
    } catch (e) { console.error("Email Error: " + e.message); }
  },

  // --- 2. MANAGER LOOKUP HELPER ---
  getManagerEmail: function(dept, type) {
    try {
      const ss = SpreadsheetApp.openById(SPREADSHEET_ID_EMAIL);
      const sheet = ss.getSheetByName("Recipients");
      if (!sheet) return "";
      
      const data = sheet.getDataRange().getValues();
      const targetDept = String(dept).trim().toLowerCase();
      
      const row = data.find(r => String(r[0]).trim().toLowerCase() === targetDept);
      if (!row) return "";
      
      // Column C (Index 2) = Manager, Column D (Index 3) = Senior
      return type === 'Senior' ? row[3] : row[2]; 
    } catch (e) {
      console.warn("Could not fetch manager email: " + e.message);
      return "";
    }
  },

  // A. APPROVED
  // A. APPROVED
  sendApprovedEmail: function(data) {
    const subject = "Holiday Request Approved: " + data.requesterName;
    const managerEmail = this.getManagerEmail(data.department, 'Manager'); 
    
    // --- AUTOMATIC BACKEND LOOKUP ENHANCEMENT ---
    // If specificDates list is missing, query the Requests DB directly via Request ID
    if ((!data.specificDates || data.specificDates.length === 0) && data.holidayId) {
      try {
        const ss = SpreadsheetApp.openById(SPREADSHEET_ID_EMAIL); 
        const reqSheet = ss.getSheetByName("Requests");
        if (reqSheet) {
          const reqIdStr = String(data.holidayId).trim();
          const finder = reqSheet.getRange("A:A").createTextFinder(reqIdStr).matchEntireCell(true); 
          const cell = finder.findNext();
          
          if (cell) {
            const rowNum = cell.getRow(); 
            const jsonDates = reqSheet.getRange(rowNum, 10).getValue(); 
            if (jsonDates) {
              data.specificDates = JSON.parse(jsonDates);
            }
          }
        }
      } catch (e) {
        console.warn("Email Service failed visual date matrix array lookup: " + e.message);
      }
    }
    // ---------------------------------------------
    
    const html = this._getTemplate_Decision({
      ...data,
      title: "Holiday Request Approved",
      status: "Approved",
      bgColor: "#f1f8e9", // Light Green
      borderColor: "#92bc20", // Wickes Green
      message: "Your holiday request has been approved."
    }); 

    this._send(data.requesterEmail, subject, html, managerEmail); 
  },

  // Template: Decision
_getTemplate_Decision: function(d) {
  // Generate a detailed weekly breakdown if specific dates are provided
  let breakdownHtml = "";
  
  if (d.specificDates && d.specificDates.length > 0) {
    // Group dates by their Week Commencing (Sunday) date
    const weeksMap = {};
    
    d.specificDates.forEach(dateStr => {
      const dateObj = new Date(dateStr);
      const dayOfWeek = dateObj.getDay();
      
      // Calculate Sunday WC date
      const wcDate = new Date(dateObj);
      wcDate.setDate(dateObj.getDate() - dayOfWeek);
      const wcIso = `${wcDate.getFullYear()}-${("0" + (wcDate.getMonth() + 1)).slice(-2)}-${("0" + wcDate.getDate()).slice(-2)}`;
      
      if (!weeksMap[wcIso]) {
        weeksMap[wcIso] = [];
      }
      weeksMap[wcIso].push(dateObj);
    });

    const sortedWcKeys = Object.keys(weeksMap).sort();
    
    // Only build a week-by-week table if the request crosses multiple weeks
    if (sortedWcKeys.length > 1 || d.specificDates.length > 5) {
      breakdownHtml = `
        <div style="margin-top: 15px; border: 1px solid #e0e0e0; border-radius: 4px; overflow: hidden;">
          <table style="width: 100%; border-collapse: collapse; font-size: 14px; text-align: left;">
            <thead>
              <tr style="background-color: #f5f5f5; border-bottom: 1px solid #e0e0e0;">
                <th style="padding: 10px; font-weight: bold; color: #073763;">Week Commencing</th>
                <th style="padding: 10px; font-weight: bold; color: #073763;">Days Booked</th>
                <th style="padding: 10px; font-weight: bold; color: #073763;">Total</th>
              </tr>
            </thead>
            <tbody>
      `;
      
      sortedWcKeys.forEach(wcKey => {
        const dateParts = wcKey.split('-');
        const formattedWc = `${dateParts[2]}/${dateParts[1]}/${dateParts[0]}`;
        
        // Sort individual days in that week and map to short format (e.g., "Fri 22/05")
        const formattedDays = weeksMap[wcKey]
          .sort((a, b) => a - b)
          .map(date => {
            const dayName = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'][date.getDay()];
            return `${dayName} ${("0" + date.getDate()).slice(-2)}/${("0" + (date.getMonth() + 1)).slice(-2)}`;
          })
          .join(', ');
        
        const weekCount = weeksMap[wcKey].length;
        
        breakdownHtml += `
          <tr style="border-bottom: 1px solid #eee;">
            <td style="padding: 10px; font-weight: bold; color: #333;">${formattedWc}</td>
            <td style="padding: 10px; color: #666; font-size: 13px;">${formattedDays}</td>
            <td style="padding: 10px; font-weight: bold; color: #333333;">${weekCount} ${weekCount === 1 ? 'Day' : 'Days'}</td>
          </tr>
        `;
      });
      
      breakdownHtml += `
            </tbody>
          </table>
        </div>
      `;
    }
  }

  const body = `
    <h2 style="color: #073763; margin-top: 0;">${d.title}</h2>
    <p>Hi ${d.requesterName},</p>
    <p>${d.message}</p>
    
    <div style="background-color: ${d.bgColor}; border-left: 6px solid ${d.borderColor}; padding: 20px; margin: 25px 0; border-radius: 4px;">
      <div style="margin-bottom: 8px;"><strong>Status:</strong> <span style="font-weight: bold; color: #333;">${d.status}</span></div>
      <div style="margin-bottom: 8px;"><strong>Overall Range:</strong> ${d.startDate} to ${d.endDate}</div>
      <div style="margin-bottom: 8px;"><strong>Total Leave Duration:</strong> ${d.totalDays} Days</div>
      ${breakdownHtml}
    </div>

    <p style="font-size: 14px; color: #666; margin-bottom: 20px;">You can view your updated balance on the portal.</p>
    
    <a href="${d.appUrl}" style="display: inline-block; padding: 12px 24px; background-color: #009ee3; color: #ffffff; text-decoration: none; border-radius: 5px; font-weight: bold; font-size: 16px;">Open Holiday Portal</a>
  `;
  return this._wrapHtml(body);
},

  const body = `

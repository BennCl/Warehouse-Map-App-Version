/**
 * TIME-LESS CONVERTER
 * Converts any date input into a pure "YYYY-MM-DD" string.
 */
const toDateStr = (input) => {
  if (!input) return "";
  const d = (input instanceof Date) ? input : new Date(input);
  if (isNaN(d.getTime())) return "";
  const y = d.getFullYear();
  const m = ("0" + (d.getMonth() + 1)).slice(-2);
  const day = ("0" + d.getDate()).slice(-2);
  return `${y}-${m}-${day}`;
};

function getInitialData() {
  try {
    const userEmail = Session.getActiveUser().getEmail();
    const ss = getSpreadsheet();
    const usersSheet = ss.getSheetByName(SHEET_NAMES.USERS);
    
    if (!usersSheet) throw new Error("Users sheet not found");

    const data = usersSheet.getDataRange().getValues();
    data.shift(); 
    
    const userRow = data.find(r => String(r[2]).toLowerCase() === userEmail.toLowerCase());
    if (!userRow) return { success: false, error: "User not found in database." };
    if (userRow[12] !== 'Active') return { success: false, error: "Account is inactive." };
    if (typeof ACCESS_LEVELS === 'undefined') throw new Error("ACCESS_LEVELS configuration missing");

    const accessCode = userRow[3];
    const accessDef = ACCESS_LEVELS[accessCode] || ACCESS_LEVELS['C3'];
    

    const startDateRaw = userRow[14];
    const startDateStr = toDateStr(startDateRaw);

    const currentYear = new Date().getFullYear();
    const nextYear = currentYear + 1;

    // --- NEW STATUTORY LOGIC ---
    const statDaysCY = Number(userRow[21]) || 0;

    // 2. Calculate Statutory Days for NY
    let statDaysNY = 0;
    const overrideNY = userRow[23]; // Check Column X for manual NY override
    if (overrideNY !== "" && overrideNY !== undefined && !isNaN(overrideNY)) {
        statDaysNY = Number(overrideNY);
    } else {
        const nyStatInfo = getVirtualStatutoryRequests_({ startDate: startDateStr, payroll: userRow[0] }, nextYear);
        statDaysNY = nyStatInfo.deduction || 0;
    }

    // 3. Setup Basic User Data
    const userData = {
      payroll: String(userRow[0]), 
      name: userRow[1], 
      email: userRow[2], 
      accessCode: accessCode, 
      rank: accessDef.rank, 
      department: userRow[5], 
      delegatedTeams: [], 
      allowance: {
        cy: { 
            total: Number(userRow[6]), 

            booked: Number(userRow[7]) + statDaysCY, 

            remaining: Number(userRow[8]) 
        }, 
        ny: { 
            total: Number(userRow[9]), 

            booked: Number(userRow[10]) + statDaysNY, 
            remaining: Number(userRow[11]) 
        } 
      },
      workPattern: userRow[13], 
      startDate: startDateStr,
      xmas: { 
          lastOff: userRow[15], 
          reserveCY: userRow[16], 
          reserveNY: userRow[17] 
      }
    };

    // 4. Fetch History (Includes Pending & Approved Requests)
    const myHistory = getHolidayHistory(userData.payroll);

    // 5. APPLY PENDING DEDUCTIONS (Visual Only)
    // We actively subtract "Pending" days from the visual balance so users don't overbook
    // while waiting for approval.
    myHistory.forEach(req => {
        if (req.status === 'Pending') {
            let rDate = new Date(req.start);
            // Safety check for invalid dates
            if (isNaN(rDate.getTime())) return;
            
            const rYear = rDate.getFullYear();
            const days = Number(req.days);

            if (rYear === currentYear) {
                // Adjust This Year (CY)
                userData.allowance.cy.booked += days;
                userData.allowance.cy.remaining -= days;
            } else if (rYear === nextYear) {
                // Adjust Next Year (NY)
                userData.allowance.ny.booked += days;
                userData.allowance.ny.remaining -= days;
            }
        }
    });


    const cyStat = getVirtualStatutoryRequests_(userData, currentYear);
    const nyStat = getVirtualStatutoryRequests_(userData, nextYear);

    // 7. Merge Virtual Requests into History
    // NEW LOGIC: Filter requests to only show PENDING items on the calendar.
// The "Approved" blocks will now come EXCLUSIVELY from the Tracker sheet (dbHistory).
const pendingOnly = myHistory.filter(r => r.status === 'Pending');

const dbHistory = getUserPublicDBEvents(userData.payroll, []); 

const combinedHistory = pendingOnly.concat(cyStat.requests, nyStat.requests, dbHistory);

    const showDashboard = accessDef.rank >= 30 || accessCode === 'Admin';
    const showTracker = accessDef.rank >= 20 || accessCode === 'Admin'; 
    const showAdminHub = accessCode === 'Admin' || accessCode === 'AdminManager';
    
    const overrides = getStatutoryOverridesCached();
    const overrideXmasGroup = String(userRow[24] || "").trim();
      const deptKeyCY = String(userRow[5] || "").trim().toLowerCase() + '_' + currentYear;
      userData.xmasGroup = overrideXmasGroup || (overrides[deptKeyCY] ? overrides[deptKeyCY].xmasGroup : "");

    return {
      success: true, 
      user: userData, 
      config: { 
        showTeamDashboard: showDashboard, 
        showTracker: showTracker, 
        showAdminHub: showAdminHub 
      }, 
      patterns: getPatternsCached(), 
      requests: combinedHistory,
      xmasExceptions: getChristmasExceptionsCached(),
      statutory: getStatutoryOverridesCached()
    };

  } catch (e) {
    console.error("getInitialData Failed: " + e.stack);
    return { success: false, error: "System Error: " + e.message };
  }
}

function getVirtualStatutoryRequests_(user, year) {
    try {

        const overrides = getStatutoryOverridesCached();
        const deptKey = String(user.department || "").trim().toLowerCase() + '_' + year;
        const rule = overrides[deptKey];

        const statItems = [
            { date: rule?.ny || `${year}-01-01`, label: "New Year's Day", type: 'H' },
            { date: rule?.xeve || `${year}-12-24`, label: "Christmas Eve (Paid Absence)", type: 'PA' },
            { date: rule?.xmas || `${year}-12-25`, label: "Christmas Day", type: 'H' },
            { date: rule?.boxing || `${year}-12-26`, label: "Boxing Day", type: 'H' }
        ];
        
        let deduction = 0;
        const validHolidays = []; 
        const validPA = [];       

        statItems.forEach(stat => {
            let isEligible = true;
            
            if (user.startDate) {
                const startObj = new Date(user.startDate);
                const statObj = new Date(stat.date);

                if (!isNaN(startObj.getTime()) && statObj < startObj) {
                    isEligible = false;
                }
            }

            if (isEligible) {
                if (stat.type === 'H') {
                    deduction += 1;
                    validHolidays.push(stat.date);
                } else if (stat.type === 'PA') {
                    // No deduction for PA
                    validPA.push(stat.date);
                }
            }
        });

        const requests = [];

        // 3. Create 'H' Block (Deducts Allowance)
        if (validHolidays.length > 0) {
            validHolidays.sort();
            requests.push({
                id: `STAT-${year}`,
                payroll: user.payroll,
                start: validHolidays[0], 
                end: validHolidays[validHolidays.length - 1], 
                days: deduction, 
                status: 'Approved', 
                source: 'Statutory', 
                specificDates: validHolidays 
            });
        }

        // 4. Create 'PA' Block (No Deduction, Different Code)
        if (validPA.length > 0) {
            validPA.sort();
            requests.push({
                id: `XEVE-${year}`,
                payroll: user.payroll,
                start: validPA[0],
                end: validPA[validPA.length - 1],
                days: 0, 
                status: 'Approved',
                source: 'Company Benefit', 
                specificDates: validPA,
                code: 'PA' 
            });
        }

        return { deduction: deduction, requests: requests };
    } catch (e) {
        console.error("Error in Virtual Stat Calc: " + e.message);
        return { deduction: 0, requests: [] }; 
    }
}

// TRACKER DATA (C5+ Access)
// ==========================================
function getTrackerData(yearInput) {
  const tStart = new Date().getTime();
  const userEmail = Session.getActiveUser().getEmail();
  
  const currentYear = new Date().getFullYear();
  const targetYear = yearInput ? Number(yearInput) : currentYear;
  const ss = getSpreadsheet();
  
  // --- FAST NATIVE DATE FORMATTER ---
  const toIso = (d) => {
      const y = d.getFullYear();
      const m = d.getMonth() + 1;
      const day = d.getDate();
      return `${y}-${m<10?'0'+m:m}-${day<10?'0'+day:day}`;
  };

  // -----------------------------------------------------------
  // 1. PRE-LOAD STATUTORY RULES (ONE READ)
  // -----------------------------------------------------------
  const overridesRaw = getStatutoryOverridesCached(); 

  // -----------------------------------------------------------
  // 2. SINGLE READ of USERS SHEET (Auth + Stat Injection)
  // -----------------------------------------------------------
  const usersSheet = ss.getSheetByName(SHEET_NAMES.USERS);
  const usersValues = usersSheet.getDataRange().getValues();
  usersValues.shift(); 
  
  const viewerRow = usersValues.find(r => String(r[2]).toLowerCase() === userEmail.toLowerCase());
  if (!viewerRow) return JSON.stringify({ success: false, error: "User not found." });

  const viewer = {
      payroll: String(viewerRow[0]),
      rank: ACCESS_LEVELS[viewerRow[3]]?.rank || 10,
      accessCode: viewerRow[3]
  };

  if (viewer.rank < 20 && viewer.accessCode !== 'Admin') {
      return JSON.stringify({ success: false, error: "🚫 Access Denied." });
  }

  const accessibleUsers = [];
  const authorizedPayrolls = new Set(); 
  const dbEvents = []; 

  usersValues.forEach(u => {
      const targetRank = ACCESS_LEVELS[u[3]]?.rank || 0;
      if (viewer.accessCode === 'Admin' || viewer.rank >= targetRank) {
          const p = String(u[0]);
          const dept = u[5];
          
          // A. Add to Accessible List
          const overrideXmasGroup = String(u[24] || "").trim();
          const deptKey = String(dept || "").trim().toLowerCase() + '_' + targetYear;
          const xGroup = overrideXmasGroup || (overridesRaw[deptKey] ? overridesRaw[deptKey].xmasGroup : "");
          
          accessibleUsers.push({
              payroll: p, name: u[1], dept: dept, pattern: u[13], xmasGroup: xGroup,
              allowance: { cy: { booked: Number(u[7])+(Number(u[21])||0), rem: u[8] }, ny: { booked: u[10], rem: u[11] } }
          });
          authorizedPayrolls.add(p);

          // B. INJECT STATUTORY DAYS (UPDATED LOGIC)
          const key = (dept || "").toLowerCase().trim() + '_' + targetYear;
          const rule = overridesRaw[key];
          
          // Define Statutory Days with DEFAULTS if override is blank
          const stats = [
            // New Year (Default Jan 1)
            { date: rule?.ny || `${targetYear}-01-01`, code: 'H' },
            // Christmas Eve (Default Dec 24) -> Shows as 'PA' (Purple)
            { date: rule?.xeve || `${targetYear}-12-24`, code: 'PA' },
            // Christmas Day (Default Dec 25)
            { date: rule?.xmas || `${targetYear}-12-25`, code: 'H' },
            // Boxing Day (Default Dec 26)
            { date: rule?.boxing || `${targetYear}-12-26`, code: 'H' }
          ];
          
          // 2. Check Start Date Eligibility
          const rawStart = u[14];
          let startTs = 0;
          // Default to epoch if missing
          if (rawStart instanceof Date) startTs = rawStart.getTime();
          else if (typeof rawStart === 'string' && rawStart) startTs = new Date(rawStart).getTime();

          stats.forEach(item => {
              // Create date from YYYY-MM-DD string
              const dTime = new Date(item.date).getTime();
              
              // Only inject if user was employed on that date
              if (dTime >= startTs) {
                  dbEvents.push({
                      payroll: p,
                      date: item.date, 
                      code: item.code 
                  });
              }
          });
      }
  });

  // -----------------------------------------------------------
  // 3. SINGLE READ of REQUESTS
  // -----------------------------------------------------------
  const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
  const reqValues = reqSheet.getDataRange().getValues();
  reqValues.shift(); 

  const requests = [];
  
  for (let i = 0; i < reqValues.length; i++) {
      const r = reqValues[i];
      const p = String(r[1]);
      
      if (!authorizedPayrolls.has(p)) continue;

      const sDate = new Date(r[3]);
      const eDate = new Date(r[4]);
      if (sDate.getFullYear() === targetYear || eDate.getFullYear() === targetYear) {
          requests.push({
              payroll: p,
              start: toIso(sDate),
              end: toIso(eDate),
              status: r[6],
              json: r[9],
              type: 'Request'
          });
      }
  }

  // -----------------------------------------------------------
  // 4. TURBO DB READ (Math Method)
  // -----------------------------------------------------------
  const dbSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const dbData = dbSheet.getDataRange().getValues();
  
  for (let i = 1; i < dbData.length; i++) {
    const p = String(dbData[i][1]);
    if (!authorizedPayrolls.has(p)) continue;

    const wc = dbData[i][4];
    if (!wc || !wc.getTime) continue; 

    const wcYear = wc.getFullYear();
    if (wcYear < targetYear - 1 || wcYear > targetYear) continue;

    for (let dIdx = 0; dIdx < 7; dIdx++) {
        const code = dbData[i][5 + dIdx];
        if (code && code !== '') {
            const dObj = new Date(wc);
            dObj.setDate(wc.getDate() + dIdx); 
            
            if (dObj.getFullYear() === targetYear) {
                dbEvents.push({
                    payroll: p,
                    date: toIso(dObj),
                    code: code
                });
            }
        }
    }
  }

  // 5. Archive Loading (Dynamic)
  const currentSystemYear = new Date().getFullYear();
  if (targetYear < currentSystemYear) {
      const archivedEvents = fetchArchivedEvents_(targetYear);
      const validArchive = archivedEvents.filter(e => authorizedPayrolls.has(String(e.payroll)));
      validArchive.forEach(e => dbEvents.push(e));
  }

  // 6. Capacity Rules
  const capSheet = ss.getSheetByName('Capacity_Rules');
  let capacityRules = {};
  if (capSheet) {
      const capData = capSheet.getDataRange().getValues();
      capData.shift();
      capData.forEach(row => { 
          if(row[0]) {
             capacityRules[row[0]] = [row[1], row[2], row[3], row[4], row[5], row[6], row[7]];
          }
      });
  }

  console.log(`Turbo Fetch Time: ${(new Date().getTime() - tStart)}ms`);

  return JSON.stringify({
    success: true,
    viewer: viewer,
    users: accessibleUsers,
    requests: requests,
    dbEvents: dbEvents, 
    patterns: getPatternsCached(),
    exceptions: getChristmasExceptionsCached(),
    statutory: getStatutoryOverridesCached(),
    capacity: capacityRules  
  });
}

// TEAM DASHBOARD (STRICT HIERARCHY >)
// ==========================================
function getTeamDashboardData() {
  const tStart = new Date().getTime();
  const userEmail = Session.getActiveUser().getEmail().toLowerCase();
  const ss = getSpreadsheet();
  
  const usersSheet = ss.getSheetByName(SHEET_NAMES.USERS);
  const dbSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
  
  const allUsers = usersSheet.getDataRange().getValues();
  const managerRow = allUsers.find(r => String(r[2]).toLowerCase() === userEmail);
  if (!managerRow) return { success: false, error: "User not found in database." };
  
  const managerRank = ACCESS_LEVELS[managerRow[3]]?.rank || 10;
  const isAdmin = managerRow[3] === 'Admin' || managerRow[3] === 'AdminManager';
  const teamMembers = [];
  const teamPayrolls = new Set();
  const currentYear = new Date().getFullYear();
  const overrides = getStatutoryOverridesCached();

  // 1. Build Team List
  for (let i = 1; i < allUsers.length; i++) {
    const u = allUsers[i];
    const targetRank = ACCESS_LEVELS[u[3]]?.rank || 0;
    if (isAdmin || managerRank > targetRank) {
      const p = String(u[0]);
      teamPayrolls.add(p);
      
      const overrideXmasGroup = String(u[24] || "").trim();
      const deptKeyCY = String(u[5] || "").trim().toLowerCase() + '_' + currentYear;
      const xGroup = overrideXmasGroup || (overrides[deptKeyCY] ? overrides[deptKeyCY].xmasGroup : "");

      teamMembers.push({
        payroll: p, name: u[1], department: u[5], workPattern: u[13],
        startDate: toDateStr(u[14]), // <--- FIX: Ensure Start Date is passed
        xmasGroup: xGroup,           // <--- FIX: Pass Xmas Group
        allowance: { cy: { remaining: u[8] }, ny: { remaining: u[11] } },
        xmas: { lastOff: u[15], reserveCY: u[16], reserveNY: u[17] }
      });
    }
  }

  const allHolidays = [];
  const dbData = dbSheet.getDataRange().getValues();
  const groupedRequests = {};
  
  for (let i = 1; i < dbData.length; i++) {
    const row = dbData[i];
    const p = String(row[1]);
    const rowReqId = row[0] ? String(row[0]) : `LEGACY-${p}-${toDateStr(row[4])}`; 

    if (!teamPayrolls.has(p)) continue;
    
    const wc = row[4];
    if (!(wc instanceof Date)) continue;

    for (let dIdx = 0; dIdx < 7; dIdx++) {
      const code = row[5 + dIdx];
      if (['H', 'HD', 'HD1', 'HD2'].includes(code)) {
        const d = new Date(wc.getTime());
        d.setDate(wc.getDate() + dIdx);
        
        if (d.getFullYear() === currentYear || d.getFullYear() === currentYear + 1) {
            if (!groupedRequests[rowReqId]) {
                groupedRequests[rowReqId] = { id: rowReqId, payroll: p, dates: [], totalDays: 0, source: 'DB' };
            }
            groupedRequests[rowReqId].dates.push(toDateStr(d));
            groupedRequests[rowReqId].totalDays += (code === 'H' ? 1 : 0.5);
        }
      }
    }
  }

  Object.values(groupedRequests).forEach(group => {
      if (group.dates.length > 0) {
          group.dates.sort();
          allHolidays.push({
              id: group.id, payroll: group.payroll, start: group.dates[0], end: group.dates[group.dates.length - 1], 
              days: group.totalDays, status: 'Approved', source: 'DB', specificDates: group.dates
          });
      }
  });

  const reqData = reqSheet.getDataRange().getValues();
  for (let i = 1; i < reqData.length; i++) {
    const row = reqData[i];
    const p = String(row[1]);
    if (!teamPayrolls.has(p)) continue;
    
    if (['Pending', 'Rejected (Cap)'].includes(row[6])) {
      let specificDates = [];
      try { if (row[9]) specificDates = JSON.parse(row[9]); } catch(e) {}
      allHolidays.push({
        id: row[0], payroll: p, name: row[2], start: toDateStr(row[3]), end: toDateStr(row[4]),
        days: row[5], status: row[6], source: 'Request', specificDates: specificDates
      });
    }
  }

  // --- FIX: INJECT VIRTUAL STATUTORY DAYS FOR TEAM ---
  teamMembers.forEach(member => {
      const cyStat = getVirtualStatutoryRequests_(member, currentYear);
      const nyStat = getVirtualStatutoryRequests_(member, currentYear + 1);
      allHolidays.push(...cyStat.requests);
      allHolidays.push(...nyStat.requests);
  });

  console.log(`Live Dashboard Load Time: ${new Date().getTime() - tStart}ms`);
  return { success: true, team: teamMembers, requests: allHolidays };
}

// --- HELPER: SMART CHAIN CALCULATOR ---
function calculateSmartChain_(payroll, newDates, patternCode, dept, allRequests) {
    if (!newDates || newDates.length === 0) return 0;

    // 1. Helper for consistent dates (Safe ISO)
    const toIso = (d) => {
        const date = new Date(d);
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        return `${year}-${month}-${day}`;
    };

    // 2. Build Master Set of Booked Dates (New + Existing Pending/Approved)
    const bookedSet = new Set(newDates.map(d => toIso(d)));
    
    for (let i = 1; i < allRequests.length; i++) {
        const r = allRequests[i];
        // Check Payroll matches & Status is valid
        if (String(r[1]) === String(payroll) && (r[6] === 'Approved' || r[6] === 'Pending')) {
            try {
                const rDates = JSON.parse(r[9]);
                rDates.forEach(d => bookedSet.add(toIso(d)));
            } catch(e) {}
        }
    }

    // 3. Define the Walker
    function walk(startIso, direction) {
        let count = 0;
        let curr = new Date(startIso);
        let gapCounter = 0;

        while(true) {
            curr.setDate(curr.getDate() + direction);
            const iso = toIso(curr);

            if (bookedSet.has(iso)) {
                count++;
                gapCounter = 0; 
                continue;
            }

            const isWork = filterWorkingDays([curr], patternCode, dept).length > 0;
            if (isWork) {
                break; 
            }

            gapCounter++;
            if (gapCounter > 14) break; 
        }
        return count;
    }

    let maxChain = 0;
    
    newDates.forEach(d => {
        const iso = toIso(d);
        // Chain = Days Behind + Days Ahead + The Day Itself (1)
        const currentChain = walk(iso, -1) + walk(iso, 1) + 1;
        if (currentChain > maxChain) maxChain = currentChain;
    });

    return maxChain;
}

function attemptCancellation(payload) {
  const lock = LockService.getScriptLock();
  try {
    if (!lock.tryLock(10000)) return { success: false, error: "System busy. Please try again" };
    if (!payload || !payload.reqId) return { success: false, error: "Missing Request ID." };

    const reqId = String(payload.reqId).trim();
    const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
    console.log(`Starting Cancellation for: ${reqId} by ${actorEmail}`);

    const ss = getSpreadsheet();
    const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
    const appSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
    const userSheet = ss.getSheetByName(SHEET_NAMES.USERS);

    // --------------------------------------------------------------------
    // 1. USE TEXTFINDER TO LOCATE THE REQUEST
    // --------------------------------------------------------------------
    const finder = reqSheet.getRange("A:A").createTextFinder(reqId).matchEntireCell(true);
    const result = finder.findNext();

    if (!result) return { success: false, error: `ID ${reqId} not found.` };

    const rowNum = result.getRow();
    const targetPayroll = String(reqSheet.getRange(rowNum, 2).getValue());
    const currentStatus = reqSheet.getRange(rowNum, 7).getValue();

    // --------------------------------------------------------------------
    // 2. STRICT SECURITY CHECK (THE "BOUNCER")
    // --------------------------------------------------------------------
    const usersData = userSheet.getDataRange().getValues();
    const actorRow = usersData.find(r => String(r[2]).toLowerCase() === actorEmail);
    const targetRow = usersData.find(r => String(r[0]) === targetPayroll);

    if (!actorRow || !targetRow) return { success: false, error: "Security Check Failed: User data not found." };

    const actorRank = ACCESS_LEVELS[actorRow[3]]?.rank || 10;
    const targetRank = ACCESS_LEVELS[targetRow[3]]?.rank || 10;
    const isAdmin = actorRow[3] === 'Admin' || actorRow[3] === 'AdminManager';
    const isSelf = String(actorRow[0]) === targetPayroll;

    if (!isSelf && !isAdmin && actorRank <= targetRank) {
        console.warn(`SECURITY BLOCK: ${actorEmail} (Rank ${actorRank}) attempted to cancel ${targetPayroll} (Rank ${targetRank})`);
        return { success: false, error: "Permission denied: You cannot cancel holidays for this colleague." };
    }

    // --- NEW: BACKEND PAST-DATE SECURITY SHIELD ---
    const reqStart = new Date(reqSheet.getRange(rowNum, 4).getValue()); // Col D = Start Date
    const todayDate = new Date();
    todayDate.setHours(0,0,0,0);
    
    let canCancel = false;
    if (isAdmin || actorRank >= 30) {
        const startOfWeek = new Date(todayDate);
        startOfWeek.setDate(todayDate.getDate() - todayDate.getDay());
        canCancel = reqStart >= startOfWeek;
    } else {
        canCancel = reqStart > todayDate;
    }
    
    if (!canCancel && !isAdmin) {
        return { success: false, error: "Security Block: This holiday has already started or passed. Please refresh your portal." };
    }

    // --------------------------------------------------------------------
    // 3. CHECK STATUS & EXECUTE
    // --------------------------------------------------------------------
    if (currentStatus === 'Cancelled') {
      const nextResult = finder.findNext();
      if (nextResult) {
         console.log(`Found duplicate ID at Row ${nextResult.getRow()}. Assuming this is the active one.`);
         return processCancellation_(ss, reqSheet, appSheet, nextResult.getRow(), reqId, targetPayroll);
      }
      return { success: false, error: "Request is already cancelled." };
    }

    return processCancellation_(ss, reqSheet, appSheet, rowNum, reqId, targetPayroll);

  } catch (err) {
    console.error("Cancellation Error: " + err.stack);
    return { success: false, error: "Error: " + err.message };
  } finally {
    lock.releaseLock();
  }
}


function processCancellation_(ss, reqSheet, appSheet, rowNum, reqId, expectedPayroll) {
    const actor = Session.getActiveUser().getEmail();
    
    // Read up to column 10 to capture the JSON array
    const rowData = reqSheet.getRange(rowNum, 1, 1, 10).getValues()[0];
    const name = rowData[2];            
    const start = rowData[3];
    const end = rowData[4];             
    const days = rowData[5];

    // --- NEW: EXTRACT SPECIFIC DATES ---
    let specificDates = [];
    try { if (rowData[9]) specificDates = JSON.parse(rowData[9]); } catch(e) {}
    
    // Fallback if no JSON dates exist
    if (specificDates.length === 0 && start && end) {
        let d = new Date(start);
        let e = new Date(end);
        while (d <= e) {
            const y = d.getFullYear();
            const m = ("0" + (d.getMonth() + 1)).slice(-2);
            const day = ("0" + d.getDate()).slice(-2);
            specificDates.push(`${y}-${m}-${day}`);
            d.setDate(d.getDate() + 1);
        }
    }
    // -----------------------------------

    // Mark Request as Cancelled
    reqSheet.getRange(rowNum, 7).setValue('Cancelled');              
    reqSheet.getRange(rowNum, 8).setValue(`Cancelled by ${actor}`);

    // 4. DUPLICATE ID SHIELD
    const appFinder = appSheet.getRange("A:A").createTextFinder(reqId).matchEntireCell(true);
    let appMatch = appFinder.findNext();
    let count = 0;

    while (appMatch) {
        const appRow = appMatch.getRow();
        const rowPayroll = String(appSheet.getRange(appRow, 2).getValue()); 
        
        if (rowPayroll === expectedPayroll) {
            const weekRange = appSheet.getRange(appRow, 6, 1, 7);
            const currentValues = weekRange.getValues()[0];
            const wcDate = new Date(appSheet.getRange(appRow, 5).getValue()); // Week Commencing Date
            
            const newValues = [currentValues.map((val, dIdx) => {
                const s = String(val).toUpperCase();
                if (s === 'H' || s === 'HD' || s === 'HD1' || s === 'HD2') {
                    
                    // --- NEW: SURGICAL CANCELLATION ---
                    const colDate = new Date(wcDate);
                    colDate.setDate(wcDate.getDate() + dIdx);
                    const y = colDate.getFullYear();
                    const m = ("0" + (colDate.getMonth() + 1)).slice(-2);
                    const day = ("0" + colDate.getDate()).slice(-2);
                    const colIso = `${y}-${m}-${day}`;
                    
                    // Only cancel if this column's date belongs to this specific request
                    if (specificDates.includes(colIso)) {
                        return 'C'; 
                    }
                    // ----------------------------------
                }
                return val; 
            })];

            weekRange.setValues(newValues);

            let newTotal = 0;
            newValues[0].forEach(v => {
                 const sv = String(v).toUpperCase();
                 if(sv === 'H') newTotal += 1;
                 if(sv === 'HD' || sv === 'HD1' || sv === 'HD2') newTotal += 0.5;
            });
            
            appSheet.getRange(appRow, 13).setValue(newTotal);
            
            // --- NEW: FIX MISSING TIMESTAMP ---
            const timeStamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
            appSheet.getRange(appRow, 14).setValue(timeStamp);
            // ----------------------------------
            
            appSheet.getRange(appRow, 15).setValue(`Cancelled by ${actor}`);
            count++;
        } else {
            console.warn(`DUPLICATE ID SHIELD ACTIVATED: Prevented cancellation of row ${appRow} belonging to ${rowPayroll}.`);
        }
        appMatch = appFinder.findNext();
    }
    
    console.log(`Updated ${count} rows in Approved Sheet safely.`);

    // 5. RECALCULATE & EMAIL
    if (start) {
        const startDate = new Date(start);
        if (!isNaN(startDate.getTime())) {
            forceRecalculateUserBalance_(expectedPayroll, startDate.getFullYear());

            const userSheet = ss.getSheetByName(SHEET_NAMES.USERS);
            const uData = userSheet.getDataRange().getValues();
            const uRow = uData.find(r => String(r[0]) === expectedPayroll);

            if (uRow) {
                 try {
                    EmailService.sendCancellationEmail({
                      requesterName: name,
                      requesterEmail: uRow[2],
                      department: uRow[5],
                      startDate: formatDateUK(startDate),
                      endDate: formatDateUK(end),
                      totalDays: days,
                      appUrl: WEB_APP_URL
                    });
                 } catch(e) {}
            }
        }
    }
    
    return { success: true, message: "Request cancelled successfully." };
}

function getHolidayHistory(payroll) {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  const ss = getSpreadsheet();
  const uSheet = ss.getSheetByName(SHEET_NAMES.USERS);
  const uData = uSheet.getDataRange().getValues();
  
  // --- SERVER-SIDE SECURITY SHIELD ---
  const actorRow = uData.find(r => String(r[2]).toLowerCase() === actorEmail);
  const targetRow = uData.find(r => String(r[0]) === String(payroll));
  if (!actorRow || !targetRow) return [];
  
  const isSelf = String(actorRow[0]) === String(payroll);
  const actorRank = ACCESS_LEVELS[actorRow[3]]?.rank || 10;
  const targetRank = ACCESS_LEVELS[targetRow[3]]?.rank || 10;
  const isAdmin = ['Admin', 'AdminManager'].includes(actorRow[3]);
  
  if (!isSelf && !isAdmin && actorRank <= targetRank) return [];
  // -----------------------------------
  
  const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
  const reqData = reqSheet.getDataRange().getValues();
  const dbSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const dbData = dbSheet.getDataRange().getValues();

  const requests = [];
  const processedIds = new Set(); 

  for (let i = 1; i < reqData.length; i++) {
    const rPayroll = String(reqData[i][1]);
    const rStatus = reqData[i][6];
    const rId = String(reqData[i][0]);

    if (rPayroll === String(payroll)) {
       if (['Pending', 'Approved', 'Rejected', 'Rejected (Cap)'].includes(rStatus)) {
           const row = reqData[i];
           let specificDates = [];
           try { if (row[9]) specificDates = JSON.parse(row[9]); } catch(e) {}

           if (specificDates.length === 0 && row[3] && row[4]) {
               let d = new Date(row[3]);
               const end = new Date(row[4]);
               while(d <= end) {
                   specificDates.push(toDateStr(d));
                   d.setDate(d.getDate() + 1);
               }
           }

           requests.push({
             id: rId,
             payroll: rPayroll,
             name: row[2],
             start: toDateStr(row[3]),
             end: toDateStr(row[4]),
             days: row[5],
             status: rStatus, 
             source: 'Request',
             specificDates: specificDates
           });
           processedIds.add(rId);
       }
    }
  }

  const dbGroups = {};
  for (let i = 1; i < dbData.length; i++) {
      const row = dbData[i];
      const p = String(row[1]);
      const rId = row[0] ? String(row[0]) : `MANUAL-${toDateStr(row[4])}`; 

      if (p === String(payroll) && !processedIds.has(rId)) { 
          const wc = row[4];
          if (wc instanceof Date) {
              for (let d = 0; d < 7; d++) {
                  const code = row[5+d];
                  if (['H', 'HD', 'HD1', 'HD2'].includes(code)) {
                      if (!dbGroups[rId]) {
                          dbGroups[rId] = { dates: [], days: 0 };
                      }
                      const dateObj = new Date(wc);
                      dateObj.setDate(wc.getDate() + d);
                      dbGroups[rId].dates.push(toDateStr(dateObj));
                      dbGroups[rId].days += (code === 'H' ? 1 : 0.5);
                  }
              }
          }
      }
  }

  Object.keys(dbGroups).forEach(id => {
      const group = dbGroups[id];
      group.dates.sort();
      requests.push({
          id: id,
          payroll: payroll,
          name: 'Manual Booking', 
          start: group.dates[0],
          end: group.dates[group.dates.length - 1],
          days: group.days,
          status: 'Approved',
          source: 'Admin Log',
          specificDates: group.dates
      });
  });

  return requests;
}

function getWeeklyDataForTeam(payrolls) {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  const ss = getSpreadsheet();
  
  // --- SERVER-SIDE SECURITY SHIELD ---
  const usersData = ss.getSheetByName(SHEET_NAMES.USERS).getDataRange().getValues();
  const actorRow = usersData.find(r => String(r[2]).toLowerCase() === actorEmail);
  if (!actorRow) return [];
  
  const actorRank = ACCESS_LEVELS[actorRow[3]]?.rank || 10;
  const isAdmin = ['Admin', 'AdminManager'].includes(actorRow[3]);

  // Strip payload down to only allowed targets
  const safePayrolls = payrolls.filter(pay => {
      const tRow = usersData.find(r => String(r[0]) === String(pay));
      if (!tRow) return false;
      const targetRank = ACCESS_LEVELS[tRow[3]]?.rank || 10;
      return isAdmin || actorRank > targetRank || String(actorRow[0]) === String(pay);
  });
  if (safePayrolls.length === 0) return [];
  // -----------------------------------

  const sheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const data = sheet.getDataRange().getValues();
  const bookings = [];
  const dayCols = [5,6,7,8,9,10,11]; 

  const userDates = {};

  for (let i = 1; i < data.length; i++) {
    const pay = String(data[i][1]);
    if (safePayrolls.includes(pay)) {
      const wcVal = data[i][4];
      if (!(wcVal instanceof Date)) continue;

      dayCols.forEach((col, idx) => {
        const val = data[i][col];
        if (['H', 'HD', 'HD1', 'HD2'].includes(val)) {
          const d = new Date(wcVal); 
          d.setDate(wcVal.getDate() + idx);
          const iso = toDateStr(d);
          
          if (!userDates[pay]) userDates[pay] = [];
          
          userDates[pay].push({ 
            iso: iso, 
            weight: (val === 'H' ? 1 : 0.5),
            code: val 
          });
        }
      });
    }
  }

  for (const [pay, dateObjs] of Object.entries(userDates)) {
    if (dateObjs.length === 0) continue;

    dateObjs.sort((a, b) => a.iso.localeCompare(b.iso));
    
    const uniqueMap = new Map();
    dateObjs.forEach(obj => uniqueMap.set(obj.iso, { weight: obj.weight, code: obj.code }));

    const uniqueDates = Array.from(uniqueMap.keys());
    
    let blockStart = uniqueDates[0];
    let prevDate = new Date(uniqueDates[0]);
    let currentBlock = [blockStart];
    let currentBlockCode = uniqueMap.get(blockStart).code;

    for (let i = 1; i < uniqueDates.length; i++) {
      const currDate = new Date(uniqueDates[i]);
      const diffDays = (currDate - prevDate) / (1000 * 60 * 60 * 24);
      const nextCode = uniqueMap.get(uniqueDates[i]).code;

      if (diffDays === 1 && currentBlockCode === nextCode && !['HD', 'HD1', 'HD2'].includes(currentBlockCode)) {
        currentBlock.push(uniqueDates[i]); 
      } else { 
        const totalDays = currentBlock.reduce((sum, dateIso) => sum + uniqueMap.get(dateIso).weight, 0);
        bookings.push({ 
            id: `DB-${pay}-${blockStart}`, 
            payroll: pay, 
            name: 'System', 
            start: blockStart, 
            end: uniqueDates[i-1], 
            days: totalDays, 
            status: 'Approved', 
            source: 'DB', 
            specificDates: currentBlock 
        });
        
        blockStart = uniqueDates[i]; 
        currentBlock = [uniqueDates[i]];
        currentBlockCode = nextCode;
      }
      prevDate = currDate;
    }

    const totalDays = currentBlock.reduce((sum, dateIso) => sum + uniqueMap.get(dateIso).weight, 0);
    bookings.push({ 
        id: `DB-${pay}-${blockStart}`, 
        payroll: pay, 
        name: 'System', 
        start: blockStart, 
        end: uniqueDates[uniqueDates.length-1], 
        days: totalDays, 
        status: 'Approved', 
        source: 'DB', 
        specificDates: currentBlock 
    });
  }
  return bookings;
}

function writeToWeeklyDB_(user, dates, val, bookedBy, requestId) {
  const ss = getSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const data = sheet.getDataRange().getValues();
  const author = bookedBy || "System";
  const weeks = {};

  dates.forEach(d => { 
      const current = new Date(d);
      const day = current.getDay(); 
      const wc = new Date(current);
      wc.setDate(current.getDate() - day); 
      wc.setHours(0, 0, 0, 0); 
      
      const k = toDateStr(wc); 
      if (!weeks[k]) weeks[k] = { wcDate: wc, days: [] }; 
      weeks[k].days.push(d); 
  });

  const dayMap = { 'Sun':5, 'Mon':6, 'Tue':7, 'Wed':8, 'Thu':9, 'Fri':10, 'Sat':11 };

  for (const [wcIso, obj] of Object.entries(weeks)) {

    let rIdx = data.findIndex(r => {
      const rowPayroll = String(r[1]);
      const rowDate = r[4];
      
      if (rowPayroll !== String(user.payroll)) return false;
      if (!(rowDate instanceof Date)) return false;
      if (toDateStr(rowDate) !== wcIso) return false;

      const rowId = String(r[0] || "").trim();
      const incomingId = String(requestId || "").trim();

      if (incomingId !== "") {
          if (rowId !== "" && rowId !== incomingId) {
              return false;
          }
      }
      
      return true;
    });

    let rowNum;

    if (rIdx === -1) {
        if (val !== '' && val !== 'C') {
            sheet.appendRow([
                requestId || `MAN-${Date.now()}`, 
                user.payroll, user.name, user.department || user.dept, 
                obj.wcDate, 
                "","","","","","","", 0, new Date(), author
            ]);
            rowNum = sheet.getLastRow();
        }
    } else {
        rowNum = rIdx + 1;
        if (requestId && val !== 'C') {
           sheet.getRange(rowNum, 1).setValue(requestId);
        }
    }

    if (rowNum) {

      obj.days.forEach(d => { 
          const dayName = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'][new Date(d).getDay()];
          sheet.getRange(rowNum, dayMap[dayName] + 1).setValue(val); 
      });

      sheet.getRange(rowNum, 5).setNumberFormat("yyyy-mm-dd");
      sheet.getRange(rowNum, 5).setValue(obj.wcDate);

      const vals = sheet.getRange(rowNum, 6, 1, 7).getValues()[0];
      let total = 0;
      vals.forEach(v => { 
          const cv = String(v).trim().toUpperCase();
          if (cv === 'H') total += 1; 
          if (cv === 'HD' || cv === 'HD1' || cv === 'HD2') total += 0.5; 
      });

      sheet.getRange(rowNum, 13).setValue(total);

      const timeStamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
      sheet.getRange(rowNum, 14).setValue(timeStamp);
      
      sheet.getRange(rowNum, 15).setValue(author);

    }
  }
}

function updateAllowance_(payroll, amount, year) {

  SpreadsheetApp.flush();
  const ss = getSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.USERS);
  const data = sheet.getDataRange().getValues();
  const idx = data.findIndex(r => String(r[0]) === String(payroll));

  if (idx === -1) {
    console.error("updateAllowance: User not found " + payroll);
    return;
  }

  const curYear = new Date().getFullYear();
  const targetYear = Number(year);
  const isCy = (targetYear === curYear);

  const colTotal = isCy ? 6 : 9;    
  const colBooked = isCy ? 7 : 10;  
  const colRem = isCy ? 8 : 11;    
  const colReserve = isCy ? 16 : 17;
  
  let statDays = Number(data[idx][21]);
  const overrideCY = data[idx][22]; 
  const overrideNY = data[idx][23]; 

  if (isCy && overrideCY !== "" && overrideCY !== undefined && !isNaN(overrideCY)) {
      statDays = Number(overrideCY);
  } else if (!isCy && overrideNY !== "" && overrideNY !== undefined && !isNaN(overrideNY)) {
      statDays = Number(overrideNY);
  } else {
      if (isNaN(statDays) || data[idx][21] === "") {
          const userObj = { payroll: data[idx][0], startDate: data[idx][14], department: data[idx][5] };
          const virtualStat = getVirtualStatutoryRequests_(userObj, targetYear);
          statDays = virtualStat.deduction || 0;
          console.log(`Auto-Corrected Stat Days for ${payroll}: ${statDays}`);
      }
  }

  const currentBooked = Number(data[idx][colBooked]) || 0;
  const total = Number(data[idx][colTotal]) || 0;
  const reserve = Number(data[idx][colReserve]) || 0;

  let newBooked = currentBooked + Number(amount);
  
  if (newBooked < 0) newBooked = 0;

function filterWorkingDays(dates, pCode, dept, explicitXmasGroup) {
  const pat = getPatternsCached()[pCode];
  if (!pat) return [];

  const safeEpoch = new Date(EPOCH_DATE);
  safeEpoch.setUTCHours(12, 0, 0, 0);
  
  const xmasExceptions = getChristmasExceptionsCached();
  const overrides = getStatutoryOverridesCached();
  
  return dates.filter(d => {
    const m = d.getMonth();
    const day = d.getDate();
    const year = d.getFullYear();
    const iso = toDateStr(d);
    
    // --- NEW XMAS ROTA LOGIC ---
    const isFestiveWindow = (m === 11 && day >= 20) || (m === 0 && day <= 4);

    if (isFestiveWindow) {
        let rotaYear = year;
        if (m === 0 && day <= 4) rotaYear = year - 1; 

        // 1. Check for the Personal Override first
        let xmasGroup = explicitXmasGroup || "";
        
        // 2. Fallback to Department matching if no personal override exists
        if (!xmasGroup) {
            const cleanDept = String(dept || "").trim().toLowerCase();
            const keysToTry = [`${cleanDept}_${rotaYear}`, `${cleanDept}_${year}`, `${cleanDept}_${new Date().getFullYear()}`];
            for (let k of keysToTry) {
                if (overrides[k] && overrides[k].xmasGroup) {
                    xmasGroup = overrides[k].xmasGroup;
                    break;
                }
            }
        }
        
        // 3. Apply the rules if a group was found
        if (xmasGroup) {
            let exceptionResult = undefined;
            const strictKey = `${rotaYear}_${xmasGroup}`;

            if (xmasExceptions[strictKey] && xmasExceptions[strictKey][iso] !== undefined) {
                exceptionResult = xmasExceptions[strictKey][iso];
            } else {
                for (let key in xmasExceptions) {
                    if (key.includes(xmasGroup) && xmasExceptions[key][iso] !== undefined) {
                        exceptionResult = xmasExceptions[key][iso];
                        break;
                    }
                }
            }

            if (exceptionResult !== undefined) {
                return exceptionResult; // true = Working Day, false = Rest Day
            }
        }
    }

    // --- STANDARD PATTERN LOGIC (If not festive, or no festive rules apply) ---
    const safeDate = new Date(d);
    safeDate.setUTCHours(12, 0, 0, 0);
    const diffTime = safeDate.getTime() - safeEpoch.getTime();
    const diffDays = Math.round(diffTime / (1000 * 60 * 60 * 24));
    
    const cycleIndex = Math.floor(diffDays / 7) % pat.cycleLength;
    const normalizedIndex = cycleIndex >= 0 ? cycleIndex : cycleIndex + pat.cycleLength;
    
    const dayName = Utilities.formatDate(d, Session.getScriptTimeZone(), "EEE");
    return pat.weeks[normalizedIndex] && pat.weeks[normalizedIndex].includes(dayName);
  });
}

    const archiveRows = [];

    const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
    if (reqSheet) {
        const reqData = reqSheet.getDataRange().getValues();
        // Loop backwards to safely delete rows without shifting indices
        for (let i = reqData.length - 1; i > 0; i--) {
            if (String(reqData[i][1]).trim() === String(payroll).trim()) {
                archiveRows.push([archiveDate, payroll, name, "Requests", JSON.stringify(reqData[i])]);
                reqSheet.deleteRow(i + 1);
            }
        }
    }

    // 2. Process 'Approved' (Tracker) Tab
    const appSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
    if (appSheet) {
        const appData = appSheet.getDataRange().getValues();
        // Loop backwards to safely delete rows
        for (let i = appData.length - 1; i > 0; i--) {
            if (String(appData[i][1]).trim() === String(payroll).trim()) {
                archiveRows.push([archiveDate, payroll, name, "Approved", JSON.stringify(appData[i])]);
                appSheet.deleteRow(i + 1);
            }
        }
    }

    // 3. Write to Archive Tab
    if (archiveRows.length > 0) {
        archiveSheet.getRange(archiveSheet.getLastRow() + 1, 1, archiveRows.length, 5).setValues(archiveRows);

} else {
    // Less than 14 Days: Hard Reject
    reqSheet.getRange(i + 1, 7).setValue('Rejected');
    reqDataForEmail.rejectionReason = "Auto-Rejected: Department Capacity Full";
    EmailService.sendRejectedEmail(reqDataForEmail);
    console.log(`-> Hard Rejected ${reqId}`);
}
      }
    }

  } catch (e) {
    console.error("Auto-Process Error: " + e.message);
  } finally {
    lock.releaseLock();
  }
}

// FIXED CAPACITY CHECK (ROBUST DATE, ID MATCHING & SHARED POOLS)
// =====================================================================================================================================================================================================================================
function checkCapacity_(dates, dept, excludePayroll = null, requestType = 'H') {
  SpreadsheetApp.flush();
  const ss = getSpreadsheet();
  const capSheet = ss.getSheetByName(SHEET_NAMES.CAPACITY);
  const dbSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  
  const capData = capSheet.getDataRange().getValues();
  const dbData = dbSheet.getDataRange().getValues();
  const normalize = (s) => String(s || "").toLowerCase().replace(/[^a-z0-9]/g, "");
  
  const requesterDeptClean = normalize(dept);
  const limitMap = [1, 2, 3, 4, 5, 6, 7]; 
  const issues = [];
  const parseAnyDate = (v) => {
      if (!v) return null;
      if (v instanceof Date) return new Date(v.getTime()); 
      if (typeof v === 'string') {
          let d = new Date(v);
          if (!isNaN(d.getTime())) return d;
          const parts = v.split('-');
          if (parts.length === 3) return new Date(parts[0], parts[1]-1, parts[2]);
      }
      return null;
  };

  // --- 2. DEFINE RULES (Integrated limits and hard blocks) ---
  const getChecks = () => {
    const d = requesterDeptClean;

    const poolMcfcAm = ['mcfcam', 'mcfcadminam'];
    const poolMcfcPm = ['mcfcpm', 'mcfcadminpm'];

    // Admin checks their own strict cap of 1, AND checks the wider shift capacity
    if (d === 'mcfcadminam') {
        return [
            { target: 'mcfcadminam', type: 'fixed', limit: 1 },
            { target: poolMcfcAm, limitSource: 'mcfcam' }
        ];
    }
    if (d === 'mcfcadminpm') {
        return [
            { target: 'mcfcadminpm', type: 'fixed', limit: 1 },
            { target: poolMcfcPm, limitSource: 'mcfcpm' }
        ];
    }

    // Main shift colleagues also check the combined pool to account for Admins who are off
    if (d === 'mcfcam') return [{ target: poolMcfcAm, limitSource: 'mcfcam' }];
    if (d === 'mcfcpm') return [{ target: poolMcfcPm, limitSource: 'mcfcpm' }];

    // --- TRANSPORT RULES ---
    // 1. Line Managers check other Line Managers AND their specific Admin Group
    // --- TRANSPORT RULES ---
    // 1. Line Managers check other Line Managers AND their specific Admin Group
    if (d === 'transportlinemanageram') return [{target:'transportlinemanageram'}, {target:'transportadmingroup1'}, {target:'transportlinemanagerpm'}, {target:'transportlinemanagernt'}];
    if (d === 'transportlinemanagerpm') return [{target:'transportlinemanagerpm'}, {target:'transportadmingroup2'}, {target:'transportlinemanageram'}, {target:'transportlinemanagernt'}];
    if (d === 'transportlinemanagernt') return [{target:'transportlinemanagernt'}, {target:'transportadmingroup3'}, {target:'transportlinemanageram'}, {target:'transportlinemanagerpm'}];

    // 2. Admin Groups check themselves AND their specific Line Manager
    if (d === 'transportadmingroup1') return [{target:'transportadmingroup1'}, {target:'transportlinemanageram'}];
    if (d === 'transportadmingroup2') return [{target:'transportadmingroup2'}, {target:'transportlinemanagerpm'}];
    if (d === 'transportadmingroup3') return [{target:'transportadmingroup3'}, {target:'transportlinemanagernt'}];

    // --- NEW: TRANSPORT SHUNTER SHARED CAPS ---
    // Groups the split departments together and points them to the original capacity limit source
    const poolShunterAM = ['transportshunteram1', 'transportshunteram2'];
    const poolShunterPM = ['transportshunterpm1', 'transportshunterpm2', 'transportshunterpm3'];
    const poolShunterNT = ['transportshunternt1', 'transportshunternt2'];

    if (poolShunterAM.includes(d)) return [{ target: poolShunterAM, limitSource: 'transportshunteram' }];
    if (poolShunterPM.includes(d)) return [{ target: poolShunterPM, limitSource: 'transportshunterpm' }];
    if (poolShunterNT.includes(d)) return [{ target: poolShunterNT, limitSource: 'transportshunternt' }];
    
    // --- COMPLIANCE RULES ---
    if (d === 'complianceanalyst') return [{ target: 'pipm1' }];
    if (d === 'pipm1') return [{ target: 'complianceanalyst', type: 'fixed', limit: 1 }, { target: 'pipm2', type: 'fixed', limit: 1 }];
    if (d === 'pipm2') {
        return [
            { target: 'pipm2' }, 
            { target: 'pipm1', type: 'fixed', limit: 1 } 
        ];
    }


    const poolAM = ['recyclingam', 'recyclingdriveram', 'recyclingteamleaderam'];
    const poolPM = ['recyclingpm', 'recyclingdriverpm', 'recyclingteamleaderpm'];


    if (d === 'recyclingsupervisoram') {
        return [
            { target: 'recyclingsupervisoram' }, 
            { target: 'recyclingsupervisorpm', type: 'fixed', limit: 1 }, 
            { target: 'recyclingteamleaderam', type: 'fixed', limit: 1 },
            { target: 'recyclinglinemanageram', type: 'fixed', limit: 1 } 
        ];
    }
    if (d === 'recyclingsupervisorpm') {
        return [
            { target: 'recyclingsupervisorpm' }, 
            { target: 'recyclingsupervisoram', type: 'fixed', limit: 1 }, 
            { target: 'recyclingteamleaderpm', type: 'fixed', limit: 1 },
            { target: 'recyclinglinemanageram', type: 'fixed', limit: 1 } 
        ];
    }


    if (d === 'recyclingteamleaderam') {
        return [
            { target: 'recyclingteamleaderam' }, 
            { target: poolAM, limitSource: 'recyclingam' }, 
            { target: 'recyclingsupervisoram', type: 'fixed', limit: 1 }, 
            { target: 'recyclingdriveram', type: 'fixed', limit: 1 }      
        ];
    }
    if (d === 'recyclingteamleaderpm') {
        return [
            { target: 'recyclingteamleaderpm' }, 
            { target: poolPM, limitSource: 'recyclingpm' }, 
            { target: 'recyclingsupervisorpm', type: 'fixed', limit: 1 }, 
            { target: 'recyclingdriverpm', type: 'fixed', limit: 1 }      
        ];
    }


    if (d === 'recyclingdriveram') {
        return [
            { target: 'recyclingdriveram' }, 
            { target: poolAM, limitSource: 'recyclingam' }, 
            { target: 'recyclingteamleaderam', type: 'fixed', limit: 1 } 
        ];
    }
    if (d === 'recyclingdriverpm') {
        return [
            { target: 'recyclingdriverpm' }, 
            { target: poolPM, limitSource: 'recyclingpm' }, 
            { target: 'recyclingteamleaderpm', type: 'fixed', limit: 1 } 
        ];
    }

    if (d === 'recyclingam') {
        return [ 
            { target: 'recyclingam', type: 'fixed', limit: 1 }, 
            { target: poolAM, limitSource: 'recyclingam' } 
        ];
    }
    if (d === 'recyclingpm') {
        return [ 
            { target: 'recyclingpm', type: 'fixed', limit: 1 },
            { target: poolPM, limitSource: 'recyclingpm' } 
        ];
    }


    return [{ target: dept }];
  };

  const checksToRun = getChecks();

  // --- 3. EXECUTE CHECKS ---
  dates.forEach(requestedDate => {
    const reqDateObj = new Date(requestedDate);
    reqDateObj.setHours(12,0,0,0); 
    
    const dayOfWeek = reqDateObj.getDay(); 
    const limitIndex = limitMap[dayOfWeek];

    const wcObj = new Date(reqDateObj);
    wcObj.setDate(reqDateObj.getDate() - dayOfWeek); 
    const wcIso = Utilities.formatDate(wcObj, Session.getScriptTimeZone(), "yyyy-MM-dd");
    
    const dbColIndex = 5 + dayOfWeek; 
    let isBlocked = false;

    for (let check of checksToRun) {
      let targetsClean = Array.isArray(check.target) ? check.target.map(normalize) : [normalize(check.target)];
      
      let maxOff;
      let maxOffHD1;
      let maxOffHD2;

      if (check.type === 'fixed') {
        maxOff = check.limit; 
        maxOffHD1 = maxOff; 
        maxOffHD2 = maxOff;
      } else {
        const limitTarget = check.limitSource ? normalize(check.limitSource) : targetsClean[0];
        const ruleRow = capData.find(r => normalize(r[0]) === limitTarget);
        if (!ruleRow) continue; 

        let useOverride = false;
        const rStart = parseAnyDate(ruleRow[8]);
        const rEnd = parseAnyDate(ruleRow[9]);
        if (rStart && rEnd) {
            const checkTime = reqDateObj.getTime();
            rStart.setHours(0,0,0,0);
            rEnd.setHours(0,0,0,0);
            if (checkTime >= rStart.getTime() && checkTime <= rEnd.getTime()) useOverride = true;
        }
        maxOff = useOverride ? ruleRow[10 + dayOfWeek] : ruleRow[limitIndex];
        maxOffHD1 = ruleRow[17] !== "" && ruleRow[17] !== undefined ? ruleRow[17] : maxOff;
        maxOffHD2 = ruleRow[18] !== "" && ruleRow[18] !== undefined ? ruleRow[18] : maxOff;
      }
      
      if (maxOff === "" || maxOff === null || maxOff === undefined) continue;
      
      let countFull = 0;
      let countHD1 = 0;
      let countHD2 = 0;
      for (let i = 1; i < dbData.length; i++) {
        const row = dbData[i]; 
        if (excludePayroll && String(row[1]) === String(excludePayroll)) continue;
        if (!targetsClean.includes(normalize(row[3]))) continue;

        const rowDate = parseAnyDate(row[4]);
        if (rowDate) {
           const rowDateIso = Utilities.formatDate(rowDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
           if (rowDateIso === wcIso) {
              const code = String(row[dbColIndex]).toUpperCase();
              if (code === 'H') countFull += 1;
              if (code === 'HD1' || code === 'HD') countHD1 += 1; 
              if (code === 'HD2') countHD2 += 1;
           }
        }
      }

      let totalHeadsOff = countFull + (countHD1 * 0.5) + (countHD2 * 0.5);
      const numMaxOff = Number(maxOff);
      const numMaxOffHD1 = Number(maxOffHD1);
      const numMaxOffHD2 = Number(maxOffHD2);

      if (requestType === 'H') {
          if (totalHeadsOff + 1 > numMaxOff) isBlocked = true;
      } else if (requestType === 'HD1') {
          if (totalHeadsOff + 0.5 > numMaxOff) isBlocked = true;
          if (countHD1 + 1 > numMaxOffHD1) isBlocked = true;
      } else if (requestType === 'HD2') {
          if (totalHeadsOff + 0.5 > numMaxOff) isBlocked = true;
          if (countHD2 + 1 > numMaxOffHD2) isBlocked = true;
      }

      if (isBlocked) break; 
    }

    if (isBlocked) issues.push(requestedDate);
  });
  
  return issues;
}

// ===========================================================================================================================================================================================
// SMART BOOKING ENGINE (With Gap Bridging)
// ===========================================================================================================================================================================================
function attemptBooking(payload) {
  const lock = LockService.getScriptLock();
  try {
    if (!lock.tryLock(10000)) return { success: false, error: "System busy. Please try again" }; 
    
    const init = getInitialData();
    if (!init.success) return init; 
    
    const ss = getSpreadsheet();
    const usersSheet = ss.getSheetByName(SHEET_NAMES.USERS);
    const usersData = usersSheet.getDataRange().getValues();

    const targetRowIndex = usersData.findIndex(r => String(r[0]) === String(payload.targetPayroll));
    if (targetRowIndex === -1) return { success: false, error: "User not found." }; 
    
    const targetRow = usersData[targetRowIndex];

    const targetUser = {
      payroll: targetRow[0], 
      name: targetRow[1], 
      rank: ACCESS_LEVELS[targetRow[3]]?.rank || 10,
      accessCode: targetRow[3], 
      department: targetRow[5], 
      workPattern: targetRow[13],
      startDate: targetRow[14], 
      xmasGroup: String(targetRow[24] || "").trim(), 
      allowance: {
        cy: { total: Number(targetRow[6]) },
        ny: { total: Number(targetRow[9]) }
      }
    };

    // --- NEW: STRICT HIERARCHY RULES ---
    const isSelf = String(init.user.payroll) === String(targetUser.payroll);
    const isSuper = init.user.accessCode === 'AdminManager';
    const isAdmin = init.user.accessCode === 'Admin';
    const isManager = init.user.rank >= 30; // C6 and above
    const isRegularUser = !isManager && !isAdmin && !isSuper; // C3, C4, C5
    const isM1Plus = init.user.rank >= 40 || isAdmin || isSuper; // M1, M2, M3, Admin, AdminManager
    // -----------------------------------

    if (!isSelf && init.user.rank <= targetUser.rank && !isAdmin && !isSuper) {
        return { success: false, error: "Permission denied." }; 
    }

    let dates = payload.dates.map(d => new Date(d)).sort((a, b) => a - b);
    if (dates.length === 0) return { success: false, error: "No dates selected." }; 

    const bookingYear = dates[0].getFullYear();
    const startYear = dates[0].getFullYear();
    const endYear = dates[dates.length-1].getFullYear();
    
    if (startYear !== endYear) return { success: false, error: "Please submit separate requests for different years" };
    
    const currentYear = new Date().getFullYear();
    if (bookingYear > currentYear + 1) return { success: false, error: `Bookings for ${bookingYear} not open.` };

    // --- MANAGER OVERRIDE CHECK ---
    let workingDates;
    if (payload.forceOverride) {
        if (!isAdmin && !isSuper) return { success: false, error: "Permission denied: Only Admins can override rest days." };
        workingDates = dates; 
        console.log(`Force Override applied for ${targetUser.payroll}`);
    } else {
        workingDates = filterWorkingDays(dates, targetUser.workPattern, targetUser.department, targetUser.xmasGroup);
    }

    if (workingDates.length === 0) return { success: false, error: "No working days selected." };

    const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
    const appSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
    const allRequestsData = reqSheet.getDataRange().getValues();
    const approvedTrackerData = appSheet.getDataRange().getValues();

    let daysToDeduct = workingDates.length;
    if (payload.halfDayType && payload.halfDayType !== "") {
        if (workingDates.length > 1) return { success: false, error: "Half Days must be booked one day at a time." }; 

        // 1. NEW: Block specific Transport drivers due to working hours laws
        const blockedTransportDepts = [
            "Transport AM", "Transport PM", "Transport NT", 
            "Transport Shunter AM", "Transport Shunter PM", "Transport Shunter NT", 
            "Transport Shunter AM1", "Transport Shunter AM2", 
            "Transport Shunter PM1", "Transport Shunter PM2", "Transport Shunter PM3", 
            "Transport Shunter NT1", "Transport Shunter NT2"
        ];
        
        if (blockedTransportDepts.includes(targetUser.department)) {
            return { success: false, error: "Transport drivers cannot book half days due to Working Time Regulations." };
        }

        // 2. FIX: C6 and M1+ can bypass the 4 half-day limit for C3/C5 colleagues
        const isManagerBooking = init.user.rank >= 30; // True if the person booking is C6 or above
        
        if (!isManagerBooking && (targetUser.accessCode === 'C3' || targetUser.accessCode === 'C5')) {
            let halfDayCount = 0;
            for (let i = 1; i < allRequestsData.length; i++) {
                const r = allRequestsData[i];
                if (String(r[1]) === String(targetUser.payroll) && ['Approved', 'Pending'].includes(r[6])) {
                    if (Number(r[5]) === 0.5) {
                        const rDate = new Date(r[3]);
                        if (rDate.getFullYear() === bookingYear) {
                            halfDayCount++;
                        }
                    }
                }
            }
            if (halfDayCount >= 4) return { success: false, error: `Half-Day Limit Reached. 4 half-days have already been booked for ${bookingYear}.` };
        }
        
        daysToDeduct = 0.5;
    }
    const trackerLookup = {};
    const dayCols = [5,6,7,8,9,10,11]; 
    
    approvedTrackerData.forEach(row => {
      if (String(row[1]) === String(targetUser.payroll)) {
        const wc = row[4];
        if (wc instanceof Date) {
          dayCols.forEach((colIdx, dIdx) => {
            const code = String(row[colIdx]).toUpperCase();
            const d = new Date(wc.getTime());
            d.setDate(wc.getDate() + dIdx);
            trackerLookup[toDateStr(d)] = code;
          });
        }
      }
    });

    const alreadyBooked = new Set();
    for (let i = 1; i < allRequestsData.length; i++) {
        if (String(allRequestsData[i][1]) === String(targetUser.payroll)) {
            const rStatus = allRequestsData[i][6];
            const rDates = JSON.parse(allRequestsData[i][9] || "[]");
            
            if (['Pending', 'Approved', 'Rejected (Cap)'].includes(rStatus)) {
                rDates.forEach(dateStr => {
                    const actualCode = trackerLookup[dateStr];
                    if (actualCode !== 'C') alreadyBooked.add(dateStr);
                });
            }
        }
    }
// =============================================================================================================================================================================================================================================================================
    // --- NEW: DUPLICATE BOOKING CHECK --- Added 01/06/2026 - check for bugs
// =============================================================================================================================================================================================================================================================================
    const hasOverlap = workingDates.some(d => alreadyBooked.has(toDateStr(d)));
    if (hasOverlap) {
        return { success: false, error: "Duplicate Request: One or more selected dates have already been requested or booked. Please view the colleague's requests and approve instead of re-booking" };
    }

    // --- 4. CAPACITY CHECK (M1+ Bypasses this) ---
   const requestCode = (payload.halfDayType && payload.halfDayType !== "") ? payload.halfDayType : 'H';
    const capacityIssues = checkCapacity_(workingDates, targetUser.department, targetUser.payroll, requestCode);
    
    let status = 'Pending'; 
    let rejectionNote = ''; 

    // C6 and below must respect capacity caps (unless an Admin forced an override)
    const shouldEnforceCap = (!isM1Plus || isSelf) && !payload.forceOverride;

    if (shouldEnforceCap && capacityIssues.length > 0) {
       const failedDatesStr = capacityIssues.map(d => toDateStr(new Date(d))).join(", ");
       const today = new Date(); today.setHours(0,0,0,0);
       const startDate = new Date(workingDates[0]); startDate.setHours(0,0,0,0);
       const diffDays = Math.ceil((startDate - today) / (1000 * 60 * 60 * 24));

       if (diffDays > 14) {
           status = 'Rejected (Cap)';
           rejectionNote = `Capacity full on: ${failedDatesStr}. Added to Waitlist.`;
       } else {
           return { success: false, error: `Request Rejected: Capacity Full on ${failedDatesStr}.` };
       }
    } 
    else if (!isRegularUser && !isSelf) {
        // C6 and M1+ automatically get 'Approved' status if cap is fine
        status = 'Approved'; 
    }
    // -----------------------------------

    if (status !== 'Rejected (Cap)') {
        
        // --- 1. EXTENDED LEAVE CHECK (M1+ Bypasses This) ---
        const chainLength = calculateSmartChain_(targetUser.payroll, workingDates, targetUser.workPattern, targetUser.department, allRequestsData);
        
        // If it's over 12 days, and the person booking is C6 or below:
        if (!isM1Plus && chainLength > 12) {
            if (!payload.comment || payload.comment.trim() === '') {
                return { 
                    requireComment: true, 
                    message: `This request forms a continuous block of ${chainLength} days.\nExtended leave requires Senior Manager approval. Please provide a reason:` 
                };
            }
            // Even if a C6 clicked "book", because it breaches 12 days, force it back to Pending so an M1 has to review it!
            status = 'Pending'; 
        }

        // --- 2. ACCRUAL COMPLIANCE (C6 and M1+ Bypass This) ---
        if (isRegularUser) {
        const accrualResult = checkAccrualCompliance_(targetUser, workingDates, allRequestsData);
        if (!accrualResult.passed) {
             return { success: false, error: `Accrual Limit Reached.\nBy ${accrualResult.periodDate}, you will only have earned ${accrualResult.limit} days.` };
        }
    }

        // --- 3. SHORT BREAK LIMIT (C6 and M1+ Bypass This) ---
        const isHalfDayBooking = payload.halfDayType && payload.halfDayType !== "";
        
        if (isRegularUser && targetUser.accessCode === 'C3' && !isHalfDayBooking) {
            const dbData = appSheet.getDataRange().getValues();
            let existingDates = [];
            for (let i = 1; i < dbData.length; i++) {
                const row = dbData[i];
                if (String(row[1]) === String(targetUser.payroll)) {
                    const wc = row[4];
                    if (wc instanceof Date) {
                        for (let d = 0; d < 7; d++) {
                            const code = row[5 + d];
                            // STRICT ISOLATION: Only count full holidays ('H') towards Short Breaks
                            if (code === 'H') {
                                const dateObj = new Date(wc);
                                dateObj.setDate(wc.getDate() + d);
                                if (dateObj.getFullYear() === bookingYear) {
                                    dateObj.setHours(12,0,0,0);
                                    existingDates.push(dateObj.getTime());
                                }
                            }
                        }
                    }
                }
            }

            const countShortBreaks = (dateArray) => {
                const uniqueSorted = [...new Set(dateArray)].sort((a, b) => a - b);
                let sbCount = 0;
                let currentRun = 0;
                for (let k = 0; k < uniqueSorted.length; k++) {
                    currentRun++;
                    const curr = uniqueSorted[k];
                    const next = uniqueSorted[k + 1];
                    let isConnected = false;
                    if (next) {
                        const diff = (next - curr) / (1000 * 60 * 60 * 24);
                        if (Math.round(diff) === 1) isConnected = true; 
                        else {
                            let gapStart = new Date(curr); gapStart.setDate(gapStart.getDate() + 1);
                            let gapIsClean = true; let safety = 0;
                            while (gapStart.getTime() < next && safety < 14) { 
                                 const check = filterWorkingDays([gapStart], targetUser.workPattern, targetUser.department);
                                 if (check.length > 0) { gapIsClean = false; break; }
                                 gapStart.setDate(gapStart.getDate() + 1); safety++;
                            }
                            if (gapIsClean && safety < 14) isConnected = true;
                        }
                    }
                    if (!isConnected) { if (currentRun <= 2) sbCount++; currentRun = 0; }
                }
                return sbCount;
            };

            const breaksBefore = countShortBreaks(existingDates);

            let newDates = [...existingDates];
            workingDates.forEach(d => {
                let temp = new Date(d);
                temp.setHours(12,0,0,0);
                newDates.push(temp.getTime());
            });
            const breaksAfter = countShortBreaks(newDates);

            if (breaksAfter > 5 && breaksAfter > breaksBefore) {
                return { success: false, error: `Short Break Limit Exceeded (${breaksAfter}/5).` };
            }
        }
    }

    const isCy = (bookingYear === currentYear);
    const reserveColIndex = isCy ? 16 : 17;
    const startReserve = Number(targetRow[reserveColIndex]) || 0;
    const isXmasDate = (d) => { const m = d.getMonth(); const dy = d.getDate();
    return (m === 11 && dy >= 23) || (m === 0 && dy <= 1); };
    const isChristmasBooking = workingDates.some(d => isXmasDate(d));

    if (isChristmasBooking) {
        const vipRoles = ['M1', 'M2', 'M3', 'AdminManager'];
        const isVip = vipRoles.includes(init.user.accessCode);
        if (isSelf && !isVip) return { success: false, error: "? Christmas dates are Manager-only." };
        usersSheet.getRange(targetRowIndex + 1, reserveColIndex + 1).setValue(0);
        usersSheet.getRange(targetRowIndex + 1, 16).setValue(bookingYear); 
        SpreadsheetApp.flush();
    }

    const totalAllow = isCy ? targetUser.allowance.cy.total : targetUser.allowance.ny.total;
    const effectiveReserve = isChristmasBooking ? 0 : startReserve;
    let liveRemaining = calculateLiveAllowance_(targetUser.payroll, bookingYear, totalAllow, effectiveReserve, allRequestsData);
    let statDeduction = 0;
    if (isCy) {
        const overrideCY = targetRow[22]; 
        statDeduction = (overrideCY !== "" && overrideCY !== undefined && !isNaN(overrideCY)) ? Number(overrideCY) : (getVirtualStatutoryRequests_(targetUser, bookingYear).deduction || 0);
    } else {
        const overrideNY = targetRow[23]; 
        statDeduction = (overrideNY !== "" && overrideNY !== undefined && !isNaN(overrideNY)) ? Number(overrideNY) : (getVirtualStatutoryRequests_(targetUser, bookingYear).deduction || 0);
    }
    liveRemaining = liveRemaining - statDeduction;

    if ((liveRemaining - daysToDeduct) < 0) {
        if (isChristmasBooking) { 
             usersSheet.getRange(targetRowIndex + 1, reserveColIndex + 1).setValue(startReserve);
             usersSheet.getRange(targetRowIndex + 1, 16).setValue(targetRow[15]); 
        }
        return { success: false, error: `Insufficient allowance. You have ${liveRemaining} days available.` }; 
    }

    let finalComment = payload.comment || '';
    if (payload.halfDayType && payload.halfDayType !== "") {
        finalComment = `[${payload.halfDayType}] ` + finalComment;
    }
    const reqId = writeToRequestsDB_(targetUser, workingDates, status, finalComment, daysToDeduct);
    if (status === 'Approved') {
        const finalCode = (payload.halfDayType && payload.halfDayType !== "") ? payload.halfDayType : 'H';
        writeToWeeklyDB_(targetUser, workingDates, finalCode, init.user.name, reqId);
        forceRecalculateUserBalance_(targetUser.payroll, bookingYear);
    }
    else if (isChristmasBooking && status === 'Pending') { 
        usersSheet.getRange(targetRowIndex + 1, reserveColIndex + 1).setValue(startReserve);
        usersSheet.getRange(targetRowIndex + 1, 16).setValue(targetRow[15]);
    }

    const emailTotalDays = payload.isHalfDay ? "0.5" : daysToDeduct.toString();
    const emailData = { 
      holidayId: reqId, requesterName: targetUser.name, requesterEmail: targetRow[2], 
      department: targetUser.department, startDate: formatDateUK(workingDates[0]), 
      endDate: formatDateUK(workingDates[workingDates.length-1]), totalDays: emailTotalDays, 
      appUrl: WEB_APP_URL, comment: payload.comment, rejectionReason: rejectionNote 
    };

    if (status === 'Approved') EmailService.sendApprovedEmail(emailData);
    else if (status === 'Rejected (Cap)') EmailService.sendRejectedCapEmail(emailData);
    else if (status === 'Rejected') EmailService.sendRejectedEmail(emailData);
    else {
        if (payload.comment && payload.comment.trim() !== '') EmailService.sendExtendedLeaveEmail(emailData);
        else EmailService.sendSubmissionConfirmation(emailData);
    } 

    let uiMessage = "Request Submitted.";
    if (status === 'Approved') uiMessage = "Request Approved.";
    else if (status === 'Rejected (Cap)') uiMessage = "Capacity Full: Added to Waitlist.";

    return { success: true, message: uiMessage };
  } catch (e) { 
    return { success: false, error: e.message };
  } finally { 
    lock.releaseLock();
  }
}

function checkAccrualCompliance_(user, newDates, allRequestsData) {
  const sortedNew = newDates.map(d => new Date(d)).sort((a,b) => a - b);
  const year = sortedNew[0].getFullYear();
  const totalAllowance = (year === new Date().getFullYear()) ? user.allowance.cy.total : user.allowance.ny.total;

  const milestones = ACCRUAL_RULES.map(rule => ({
    date: new Date(year, rule.month, rule.day, 23, 59, 59), 
    pct: rule.pct,
    limit: Math.ceil(totalAllowance * rule.pct)
  }));

  for (const milestone of milestones) {

      if (milestone.date < sortedNew[0]) continue;

      let daysUsed = 0;
      for (let i = 1; i < allRequestsData.length; i++) {
        const r = allRequestsData[i];
        if (String(r[1]) === String(user.payroll) && (r[6] === 'Approved' || r[6] === 'Pending')) {
          try {
            const rDates = JSON.parse(r[9]).map(d => new Date(d));

            const relevantDays = rDates.filter(d => d <= milestone.date && d.getFullYear() === year).length;
            daysUsed += relevantDays;
          } catch (e) {}
        }
      }

      const newDaysApplicable = sortedNew.filter(d => d <= milestone.date).length;
      
      if (newDaysApplicable === 0) continue;

      const totalAtMilestone = daysUsed + newDaysApplicable;


      if (totalAtMilestone > milestone.limit) {
          const dateString = Utilities.formatDate(milestone.date, Session.getScriptTimeZone(), "dd MMM");
          return { 
            passed: false, 
            limit: milestone.limit, 
            current: daysUsed, 
            requested: newDaysApplicable,
            periodDate: dateString, 
            percent: milestone.pct * 100
          };
      }
  }

  return { passed: true };
}

function checkWaitlistMatches_(freedDates, dept) {
  if (!freedDates || freedDates.length === 0) return;

  const ss = getSpreadsheet();
  const reqSheet = ss.getSheetByName(SHEET_NAMES.REQUESTS);
  const userSheet = ss.getSheetByName(SHEET_NAMES.USERS);

  const reqData = reqSheet.getDataRange().getValues();
  const userData = userSheet.getDataRange().getValues(); 

  const freedSet = new Set(freedDates.map(d => new Date(d).toISOString().slice(0,10)));
  const candidates = [];

  for (let i = 1; i < reqData.length; i++) {
    const r = reqData[i];
    const payroll = String(r[1]);
    const userRow = userData.find(u => String(u[0]) === payroll);

    if (!userRow || userRow[5] !== dept) continue;
    if (r[6] !== 'Rejected (Cap)') continue;

    let wantsFreedDates = false;
    let reqDateStrings = [];
    try {
        reqDateStrings = JSON.parse(r[9]);
        wantsFreedDates = reqDateStrings.some(d => freedSet.has(d));
    } catch(e) { continue; }

    if (!wantsFreedDates) continue;

    const year = new Date(reqDateStrings[0]).getFullYear();
    const curYear = new Date().getFullYear();
    const remCol = (year === curYear) ? 8 : 11; 
    const remaining = Number(userRow[remCol]);
    const needed = Number(r[5]); 

    if (remaining >= needed) {
        candidates.push({
            name: r[2],
            payroll: payroll,
            dates: `${reqDateStrings[0]} to ${reqDateStrings[reqDateStrings.length-1]}`,
            days: needed,
            remaining: remaining
        });
    }
  }

  if (candidates.length > 0) {
      EmailService.sendWaitlistAlert(dept, candidates, freedDates[0]);
  }
}

function optInForChristmas(yearType) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return { success: false, error: "System busy. Please try again" };
  
  try {
    const init = getInitialData();
    if (!init.success) return init;
    
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAMES.USERS);
    const data = sheet.getDataRange().getValues();
    const idx = data.findIndex(r => String(r[0]) === String(init.user.payroll));
    
    if (idx === -1) return { success: false, error: "User not found." };
    

    const currentSystemYear = new Date().getFullYear();
    const today = new Date();
    let targetYear, reserveColIndex, remColIndex, currentReserve;

    if (yearType === 'NY') {
        targetYear = currentSystemYear + 1;
        reserveColIndex = 17; 
        remColIndex = 11;   
        currentReserve = data[idx][17]; 
    } else {
        targetYear = currentSystemYear;
        reserveColIndex = 16; 
        remColIndex = 8;      
        currentReserve = data[idx][16];
        

        if (today.getMonth() > 9) {
             return { success: false, error: "Christmas bookings for this year are now closed." };
        }
    }

    if (String(currentReserve) === 'Refused') {
      return { success: false, error: `Your request for Christmas ${targetYear} was declined. You cannot opt-in again.` };
    }

    if (Number(currentReserve) > 0) {
      return { success: false, error: `You have already opted in for ${targetYear}.` };
    }

    const lastOff = Number(data[idx][15]); // Col P
    if (lastOff === targetYear) {
      return { success: false, error: `Christmas ${targetYear} is already confirmed.` };
    }

    if (lastOff === (targetYear - 1) || lastOff === (targetYear - 2)) {
      return { success: false, error: "Not Eligible: You have had Christmas off in the last 2 years." };
    }

    const allowanceRem = Number(data[idx][remColIndex]);
    if (allowanceRem < XMAS_CONF.reserveCost) {
      return { success: false, error: `Insufficient allowance for ${targetYear}. You need ${XMAS_CONF.reserveCost} days.` };
    }

    sheet.getRange(idx + 1, remColIndex + 1).setValue(allowanceRem - XMAS_CONF.reserveCost); 
    sheet.getRange(idx + 1, reserveColIndex + 1).setValue(XMAS_CONF.reserveCost);

    return { success: true, message: `Reserved 5 days for Christmas ${targetYear}.` };

  } catch (e) {
    return { success: false, error: e.message };
  } finally {
    lock.releaseLock();
  }
}

function optOutForChristmas(yearType) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return { success: false, error: "System busy. Please try again" };

  try {
    const init = getInitialData();
    if (!init.success) return init;

    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAMES.USERS);
    const data = sheet.getDataRange().getValues();
    const idx = data.findIndex(r => String(r[0]) === String(init.user.payroll));
    
    if (idx === -1) return { success: false, error: "User not found." };

    let reserveColIndex, remColIndex, currentReserve;

    if (yearType === 'NY') {
        reserveColIndex = 17; 
        remColIndex = 11;  
        currentReserve = Number(data[idx][17]); 
    } else {
        reserveColIndex = 16; 
        remColIndex = 8;    
        currentReserve = Number(data[idx][16]);
    }

    if (currentReserve === 0) {
      return { success: false, error: "No reserve found to refund for this year." };
    }

    const allowanceRem = Number(data[idx][remColIndex]);
    
    sheet.getRange(idx + 1, remColIndex + 1).setValue(allowanceRem + currentReserve);
    sheet.getRange(idx + 1, reserveColIndex + 1).setValue(0); 

    return { success: true, message: "Opt-out successful. Days returned." };
  } catch (e) {
    return { success: false, error: e.message };
  } finally {
    lock.releaseLock();
  }
}

function injectStatutoryHolidays_(user, year, existingHolidays) {
    const stats = [
        { date: `${year}-01-01`, name: "New Year's Day" },
        { date: `${year}-12-25`, name: "Christmas Day" },
        { date: `${year}-12-26`, name: "Boxing Day" }
    ];

    let deduction = 0;
    const validDates = [];

    stats.forEach(stat => {

        if (user.startDate) {
            let startObj;
            if (user.startDate instanceof Date) {
                startObj = user.startDate;
            } else if (typeof user.startDate === 'string' && user.startDate.includes('/')) {
                const parts = user.startDate.split('/');
                startObj = new Date(parts[2], parts[1]-1, parts[0]);
            } else {
                startObj = new Date(user.startDate);
            }
            if (new Date(stat.date) < startObj) return; 
        }
        deduction += 1; 
        validDates.push(stat.date);
    });

    const virtualHolidays = [];
    if (deduction > 0) {
        virtualHolidays.push({
            id: `STAT-${year}`,
            payroll: user.payroll,
            start: validDates[0], 
            end: validDates[validDates.length - 1], 
            days: deduction, 
            status: 'Approved', 
            source: 'Statutory', 
            specificDates: validDates 
        });
    }

    return { deduction, holidays: virtualHolidays };
}

    const usersData = uSheet.getDataRange().getValues();
    const uRow = usersData.find(r => String(r[0]) === payroll);

        requesterEmail: userObj.email,
        department: userObj.department,

  } catch (e) {

  } finally {

  }
}

    const dateObjs = dates.map(d => new Date(d));
    
    const chainLength = calculateSmartChain_(payroll, dateObjs, u[13], u[5], data);

        return { 
          success: false, 
          error: `Approval Blocked: This request forms a continuous block of ${chainLength} days. Extended leave requires M1+ approval.` 
        };
    }

    const capacityIssues = checkCapacity_(dateObjs, u[5], payroll);
    if (capacityIssues.length > 0 && !isM1Plus) {
        return { 
          success: false, 
          error: `Approval Blocked: Capacity is full. Only M1+ or Admins can override capacity limits.` 
        };
    }
    reqSheet.getRange(rowIndex + 1, 7).setValue('Approved');
    
    const userObj = { payroll: payroll, name: u[1], department: u[5] };
    
    let reqType = 'H';
    const commentText = String(row[7] || '');
    if (commentText.includes('[HD1]')) reqType = 'HD1';
    else if (commentText.includes('[HD2]')) reqType = 'HD2';
    else if (Number(row[5]) === 0.5) reqType = 'HD1'; 

    writeToWeeklyDB_(userObj, dates, reqType, Session.getActiveUser().getEmail(), reqId);

    const year = new Date(dates[0]).getFullYear();
    forceRecalculateUserBalance_(payroll, year);
    
    const emailData = {
      holidayId: reqId,
      requesterName: u[1],
      requesterEmail: u[2],
      department: u[5],
      startDate: formatDateUK(dates[0]),
      endDate: formatDateUK(dates[dates.length - 1]),
      totalDays: row[5],
      appUrl: WEB_APP_URL,
      specificDates: dates
    };
    
    EmailService.sendApprovedEmail(emailData);
    return { success: true, message: "Request approved successfully." };
    
  } catch (e) {
    return { success: false, error: e.message };
  } finally {
    lock.releaseLock();
  }
}

function calculateLiveAllowance_(payroll, year, totalAllowance, reserve, allRequestsData) {
  let usedDays = 0;

  for (let i = 1; i < allRequestsData.length; i++) {
    const r = allRequestsData[i];

    if (String(r[1]) !== String(payroll)) continue;

    if (['Approved', 'Pending'].includes(r[6])) {

       let rDate = new Date(r[3]);
       if (rDate.getFullYear() === Number(year)) {
         usedDays += Number(r[5]); 
       }
    }
  }

  return totalAllowance - usedDays - reserve;
}

function managerRefuseChristmas(payroll, yearType) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return { success: false, error: "System busy. Please try again" };

  try {
    const ss = getSpreadsheet();
    const sheet = ss.getSheetByName(SHEET_NAMES.USERS);
    const data = sheet.getDataRange().getValues();
    const idx = data.findIndex(r => String(r[0]) === String(payroll));

    if (idx === -1) return { success: false, error: "User not found." };

    // --- THE FIX: Strict hierarchy comparison ---
    const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
    const actorRow = data.find(r => String(r[2]).toLowerCase() === actorEmail);
    if (!actorRow) return { success: false, error: "User authentication failed." };

    const actorRank = ACCESS_LEVELS[actorRow[3]]?.rank || 10;
    const targetRank = ACCESS_LEVELS[data[idx][3]]?.rank || 10;
    const isAdmin = actorRow[3] === 'Admin' || actorRow[3] === 'AdminManager';

    if (!isAdmin && actorRank <= targetRank) {
        return { success: false, error: "Permission Denied: You cannot manage this colleague." };
    }
    // --------------------------------------------

    let resCol = (yearType === 'NY') ? 17 : 16;
    let remCol = (yearType === 'NY') ? 11 : 8;

    const currentReserve = data[idx][resCol]; 

    if (currentReserve === 'Refused') return { success: false, error: "Already refused." };
    if (typeof currentReserve !== 'number' || currentReserve <= 0) {
        return { success: false, error: "User has no active reserve to refuse." };
    }

    const currentRem = Number(data[idx][remCol]);
    sheet.getRange(idx + 1, remCol + 1).setValue(currentRem + currentReserve); 
    sheet.getRange(idx + 1, resCol + 1).setValue("Refused"); 

    return { success: true, message: "Christmas request refused. Days refunded." };

  } catch (e) {
    return { success: false, error: e.message };
  } finally {
    lock.releaseLock();
  }
}

function getUserPublicDBEvents(payroll, existingRequests) {
  // --- SERVER-SIDE SECURITY SHIELD ---
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  const ss = getSpreadsheet();
  const uData = ss.getSheetByName(SHEET_NAMES.USERS).getDataRange().getValues();
  const actorRow = uData.find(r => String(r[2]).toLowerCase() === actorEmail);
  const targetRow = uData.find(r => String(r[0]) === String(payroll));
  
  if (!actorRow || !targetRow) return [];
  const isSelf = String(actorRow[0]) === String(payroll);
  const actorRank = ACCESS_LEVELS[actorRow[3]]?.rank || 10;
  const targetRank = ACCESS_LEVELS[targetRow[3]]?.rank || 10;
  const isAdmin = ['Admin', 'AdminManager'].includes(actorRow[3]);
  
  if (!isSelf && !isAdmin && actorRank <= targetRank) return [];
  // -----------------------------------

  const sheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const data = sheet.getDataRange().getValues();
  
  const coveredDates = new Set();
  existingRequests.forEach(req => {
    if (req.specificDates) req.specificDates.forEach(d => coveredDates.add(d));
    else {
      let d = new Date(req.start);
      const end = new Date(req.end);
      while(d <= end) { coveredDates.add(toDateStr(d)); d.setDate(d.getDate() + 1); }
    }
  });

  const validEvents = [];
  const dayCols = [5,6,7,8,9,10,11]; 

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][1]) !== String(payroll)) continue;
    const wc = data[i][4];
    const rowReqId = data[i][0]; 

    if (!(wc instanceof Date)) continue;

    dayCols.forEach((col, idx) => {
      const code = data[i][col];
      if (!code || code === '' || HIDE_FROM_DASHBOARD.includes(code)) return;

      const d = new Date(wc); 
      d.setDate(wc.getDate() + idx);
      const iso = toDateStr(d)

      if (coveredDates.has(iso) && ['H','HD','HD1','HD2'].includes(code)) return;
      validEvents.push({ date: iso, code: code, reqId: rowReqId }); 
    });
  }

  if (validEvents.length === 0) return [];
  validEvents.sort((a,b) => a.date.localeCompare(b.date));
  
  const blocks = [];
  let currentBlock = [validEvents[0].date];
  let currentCode = validEvents[0].code;
  let currentId = validEvents[0].reqId; 
  let prevDate = new Date(validEvents[0].date);

  for (let i = 1; i < validEvents.length; i++) {
    const item = validEvents[i];
    const thisDate = new Date(item.date);
    const diff = (thisDate - prevDate) / (1000 * 60 * 60 * 24);
    const sameId = String(item.reqId) === String(currentId);
    const isExplicitRequest = sameId && String(currentId).startsWith('REQ-');
    
    if (item.code === currentCode && ((diff === 1 && sameId) || isExplicitRequest)) {
      currentBlock.push(item.date);
    } else {
      blocks.push({
        id: currentId || `DB-${payroll}-${currentBlock[0]}`, 
        payroll: payroll,
        name: currentCode === 'H' ? 'Manual Booking' : (TOOLTIPS[currentCode] || currentCode),
        code: currentCode, 
        start: currentBlock[0],
        end: currentBlock[currentBlock.length - 1],
        days: (['HD','HD1','HD2'].includes(currentCode)) ? currentBlock.length * 0.5 : currentBlock.length,
        status: 'Approved',
        source: 'Admin Log',
        specificDates: currentBlock
      });
      currentBlock = [item.date];
      currentCode = item.code;
      currentId = item.reqId; 
    }
    prevDate = thisDate;
  }

  blocks.push({
        id: currentId || `DB-${payroll}-${currentBlock[0]}`, 
        payroll: payroll,
        name: currentCode === 'H' ? 'Manual Booking' : (TOOLTIPS[currentCode] || currentCode),
        code: currentCode, 
        start: currentBlock[0],
        end: currentBlock[currentBlock.length - 1],
        days: (['HD','HD1','HD2'].includes(currentCode)) ? currentBlock.length * 0.5 : currentBlock.length,
        status: 'Approved',
        source: 'Admin Log',
        specificDates: currentBlock
  });
  
  return blocks;
}

function columnLetterToIndex_(letter) {
  let column = 0;
  let length = letter.length;
  for (let i = 0; i < length; i++) {
    column += (letter.charCodeAt(i) - 64) * Math.pow(26, length - i - 1);
  }
  return column - 1; 
}

function archiveYearToDrive(targetYear) {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  if (!ADMIN_WHITELIST.map(e => e.toLowerCase()).includes(actorEmail)) throw new Error("Security Alert: Unauthorised access attempt blocked.");

  const startTime = new Date().getTime();
  console.log(`Starting Archive for ${targetYear}...`);
  
  const ss = getSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  const data = sheet.getDataRange().getValues();
  const headers = data.shift(); 

  let currentArchive = fetchArchivedEvents_(targetYear) || [];
  console.log(`Loaded ${currentArchive.length} existing events from JSON.`);

  const rowsToKeep = [headers]; 
  const dayCols = [5, 6, 7, 8, 9, 10, 11]; 
  let archivedCount = 0;
  let isComplete = true;

  for (let r = 0; r < data.length; r++) {
    const row = data[r];

    if (new Date().getTime() - startTime > MAX_RUNTIME_MS) {
      console.warn("Time limit reached. Pausing archive job.");
      for (let rest = r; rest < data.length; rest++) {
        rowsToKeep.push(data[rest]);
      }
      isComplete = false;
      break; 
    }
    // ----------------------

    const wcDate = row[4];
    let shouldArchive = false;

    if (wcDate instanceof Date) {
      if (wcDate.getFullYear() === targetYear) {
        shouldArchive = true;
        const payroll = String(row[1]);
        dayCols.forEach((colIndex, i) => {
          const code = row[colIndex];
          if (code && code !== "") {
            const d = new Date(wcDate);
            d.setDate(wcDate.getDate() + i);

            currentArchive.push({
              p: payroll,
              d: toDateStr(d),
              c: code
            });
            archivedCount++;
          }
        });
      }
    }

    if (!shouldArchive) rowsToKeep.push(row);
  }

  if (archivedCount > 0 || !isComplete) {
    const folder = getTargetArchiveFolder_();
    const fileName = `holiday_archive_${targetYear}.json`;

    const files = folder.getFilesByName(fileName);
    if (files.hasNext()) files.next().setTrashed(true);
    
    folder.createFile(fileName, JSON.stringify(currentArchive), MimeType.PLAIN_TEXT);
    console.log(`Saved Total ${currentArchive.length} events to Drive (Added ${archivedCount} this run).`);

    sheet.clearContents();
    sheet.getRange(1, 1, rowsToKeep.length, rowsToKeep[0].length).setValues(rowsToKeep);
    console.log("Sheet updated.");
  } else {
    console.log("No new data found to archive.");
  }

  if (!isComplete) {
    console.log("⚠️ JOB PAUSED: Run the function again to continue archiving.");
  } else {
    console.log("✅ JOB COMPLETE: All rows for " + targetYear + " have been archived.");
  }
}

/**
 * 2. READER FUNCTION
 */
function fetchArchivedEvents_(year) {
  try {
    const fileName = `holiday_archive_${year}.json`;
    const folder = getTargetArchiveFolder_();
    const files = folder.getFilesByName(fileName);
    
    if (!files.hasNext()) return [];
    
    const file = files.next();
    const jsonContent = file.getBlob().getDataAsString();
    const rawEvents = JSON.parse(jsonContent);
    
    return rawEvents.map(e => ({
      payroll: e.p,
      date: e.d,
      code: e.c
    }));

  } catch (e) {
    console.error("Error reading archive: " + e.message);
    return [];
  }
}

function transportLogEvent(payload) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return { success: false, error: "System busy. Please try again" };

  try {
    const init = getInitialData();
    if (!init.success) return init;
    const manager = init.user;

    const dept = manager.department ? manager.department.toLowerCase() : "";
    
    // --- THE FIX: Must be transport AND rank 30+ (Manager) ---
    const isTransportManager = (dept.includes('transport') && manager.rank >= 30) || manager.accessCode.startsWith('Admin');
    if (!isTransportManager) return { success: false, error: "⛔ Access Denied: Transport Managers only." };
    // ---------------------------------------------------------

    const ss = getSpreadsheet();
    const usersSheet = ss.getSheetByName("Users");
    const usersData = usersSheet.getDataRange().getValues();
    const targetRow = usersData.find(r => String(r[0]) === String(payload.targetPayroll));

    if (!targetRow) return { success: false, error: "Driver not found." };

    const targetUser = { payroll: targetRow[0], name: targetRow[1], department: targetRow[5] };
    const dates = payload.dates.map(d => new Date(d));

    writeToWeeklyDB_(targetUser, dates, payload.code, manager.name);

    return { success: true, message: `Logged ${payload.code} for ${targetUser.name}.` };

  } catch (e) {
    return { success: false, error: e.message };
  } finally {
    lock.releaseLock();
  }
}

// ========================================================================================================================================================================
function forceRecalculateUserBalance_(payroll, targetYear) {
  SpreadsheetApp.flush();
  
  const ss = getSpreadsheet();
  const dbSheet = ss.getSheetByName("Approved");
  const userSheet = ss.getSheetByName("Users");
  
  const dbData = dbSheet.getDataRange().getValues();
  const yearNum = Number(targetYear);

  const dayCols = [5, 6, 7, 8, 9, 10, 11];
  
  let actualBookedCount = 0;

  for (let i = 1; i < dbData.length; i++) {
    const row = dbData[i];

    if (String(row[1]) !== String(payroll)) continue; 

    const wc = row[4];
    if (!(wc instanceof Date)) continue; 

    dayCols.forEach((colIdx, dayOffset) => {
        const val = String(row[colIdx]).toUpperCase();

        if (val === 'H' || val === 'HD' || val === 'HD1' || val === 'HD2') {
            const cellDate = new Date(wc);
            cellDate.setDate(wc.getDate() + dayOffset);

            if (cellDate.getFullYear() === yearNum) {
                if (val === 'H') actualBookedCount += 1;
                if (val === 'HD' || val === 'HD1' || val === 'HD2') actualBookedCount += 0.5;
            }
        }
    });
  }

  const usersData = userSheet.getDataRange().getValues();
  const userIdx = usersData.findIndex(r => String(r[0]) === String(payroll));
  
  if (userIdx === -1) {
    console.error("Recalc failed: User not found " + payroll);
    return;
  }

  const curYear = new Date().getFullYear();
  const isCy = (yearNum === curYear);

  const colBooked = isCy ? 7 : 10; 
  const colRem = isCy ? 8 : 11;   
  const colReserve = isCy ? 16 : 17; 

  const total = Number(usersData[userIdx][isCy ? 6 : 9]) || 0; 
  const reserve = Number(usersData[userIdx][colReserve]) || 0;
  const statDays = Number(usersData[userIdx][21]) || 0;     

  const trueRemaining = total - actualBookedCount - reserve - statDays;

  console.log(`Recalc for ${payroll} (${yearNum}): Booked=${actualBookedCount}, Rem=${trueRemaining}`);

  userSheet.getRange(userIdx + 1, colBooked + 1).setValue(actualBookedCount);
  userSheet.getRange(userIdx + 1, colRem + 1).setValue(trueRemaining);

  SpreadsheetApp.flush(); 
}

function sendHolidayNudges() {

  const today = new Date();
  const month = today.getMonth(); 

  let targetBooked = 0;
  let periodName = "";

  if (month === 2) { targetBooked = 5; periodName = "end of March"; }    
  else if (month === 5) { targetBooked = 10; periodName = "end of June"; }  
  else if (month === 8) { targetBooked = 15; periodName = "end of September"; } 
  else {
      console.log("Not a nudge month (Mar, Jun, Sept). Skipping.");
      return; 
  }

  console.log(`Starting Nudge Run: Target is ${targetBooked} days by ${periodName}`);

  const ss = getSpreadsheet();
  const usersSheet = ss.getSheetByName("Users");
  const data = usersSheet.getDataRange().getValues(); 

  let sentCount = 0;
  for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row[12] !== 'Active') continue; 
      
      const email = row[2];
      const name = row[1];
      const bookedCY = Number(row[7]); 

      if (bookedCY < targetBooked) {
          try {
              EmailService.sendNudgeEmail({
                  name: name,
                  email: email,
                  booked: bookedCY,
                  target: targetBooked,
                  period: periodName
              });
              sentCount++;
              console.log(`Nudged ${name} (${bookedCY}/${targetBooked})`);
          } catch(e) {
              console.error(`Failed to nudge ${name}: ${e.message}`);
          }
      }
  }
  console.log(`Nudge Run Complete. Sent ${sentCount} emails.`);
}

  const data = sourceSheet.getRange(START_ROW, 1, TOTAL_ROWS, TOTAL_COLS).getDisplayValues();
  const daysMap = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  
  const output = [];

  data.forEach(row => {
    const driverName = row[NAME_COL_INDEX];
    if (!driverName || driverName.toLowerCase() === 'names') return;

    const weeks = [];
    for (let w = 0; w < WEEKS_IN_ROTATION; w++) {
      let workingDays = [];
      for (let d = 0; d < 7; d++) {
        const idx = (SHIFT_DATA_START_COL - 1) + (w * 7) + d;
        if(row[idx]) {
            const val = String(row[idx]).trim().toUpperCase();
            if (val.includes('W') && !val.includes('RD')) workingDays.push(daysMap[d]);
        }
      }
      weeks.push(workingDays.join(", "));
    }
    
    const fingerprint = JSON.stringify(weeks);
    let matchedCode = patternMap.get(fingerprint) || "NO MATCH";
    
    output.push([driverName, matchedCode]);
  });

  let mapSheet = targetSS.getSheetByName('Recycling_Mapping');
  if (!mapSheet) mapSheet = targetSS.insertSheet('Recycling_Mapping');
  else mapSheet.clear();
  
  mapSheet.appendRow(["Driver Name", "Pattern Code"]);
  
  if (output.length > 0) {
    mapSheet.getRange(2, 1, output.length, 2).setValues(output);
    mapSheet.getRange(1, 1, 1, 2).setFontWeight("bold").setBackground("#f6ffed"); 
    mapSheet.autoResizeColumns(1, 2);
  }
  
  console.log(`Mapped ${output.length} Recycling drivers.`);
}

function runHolidayComparisonReport() {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();

  const appData = appSheet.getDataRange().getValues();
  
  const trackerIndex = new Set();
  const dayCols = [5, 6, 7, 8, 9, 10, 11];
  
  for (let i = 1; i < appData.length; i++) {
    const payroll = String(appData[i][1]);
    const wc = appData[i][4];

        const d = new Date(wc);
        d.setDate(wc.getDate() + dIdx);
        const iso = toDateStr(d)
        trackerIndex.add(`${payroll}_${iso}`);
      }
    });
  }

  const discrepancies = [["Payroll", "Name", "Date Missing", "Request ID", "Status"]];
  
  for (let i = 1; i < reqData.length; i++) {
    const reqId = reqData[i][0];
    const payroll = String(reqData[i][1]);
    const name = reqData[i][2];
    const status = reqData[i][6];
    const jsonDates = reqData[i][9];

    if (status === 'Approved' && jsonDates) {
      try {
        const dates = JSON.parse(jsonDates);
        dates.forEach(dateStr => {
          if (!trackerIndex.has(`${payroll}_${dateStr}`)) {
            discrepancies.push([payroll, name, dateStr, reqId, status]);
          }
        });
      } catch (e) {}
    }
  }

  let reportSheet = ss.getSheetByName("RECONCILIATION_REPORT");
  if (reportSheet) {
    reportSheet.clear();
  } else {
    reportSheet = ss.insertSheet("RECONCILIATION_REPORT");
  }
  
  if (discrepancies.length > 1) {
    reportSheet.getRange(1, 1, discrepancies.length, discrepancies[0].length).setValues(discrepancies);
    reportSheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#d1e7dd");

  } else {
    console.log("No discrepancies found! Both sources match.");
    reportSheet.getRange(1,1).setValue("No missing dates found. All Approved requests match the Tracker.");
  }
}

 * Reads the 'RECONCILIATION_REPORT' tab.
 * Writes 'H' to the Approved tab for every row present.
 */
// =================================================================================================================================================================================================
function applyFilteredReconciliation() {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  if (!ADMIN_WHITELIST.map(e => e.toLowerCase()).includes(actorEmail)) throw new Error("Security Alert: Unauthorised access attempt blocked.");

  const ss = SpreadsheetApp.openById('1wLTDovDcceXdVEK4H0R-2L4OkrAgW8f61pKYYWIPCA4');
  const reportSheet = ss.getSheetByName("RECONCILIATION_REPORT");
  const appSheet = ss.getSheetByName("Approved");
  const userSheet = ss.getSheetByName("Users"); 
  
  // Dynamiczne pobieranie wszystkich wierszy
  const lastRow = reportSheet.getLastRow();
  if (lastRow < 2) return;
  const reportData = reportSheet.getRange(2, 1, lastRow - 1, 5).getValues(); 
  
  const appValues = appSheet.getDataRange().getValues();
  const userValues = userSheet.getDataRange().getValues();
  
  const colMap = { 0: 6, 1: 7, 2: 8, 3: 9, 4: 10, 5: 11, 6: 12 };
  let count = 0;
  const uniquePayrolls = new Set();

  // Helper to format dates correctly
  const toDateStr = (input) => {
    if (!input) return "";
    const d = (input instanceof Date) ? input : new Date(input);
    if (isNaN(d.getTime())) return "";
    const y = d.getFullYear();
    const m = ("0" + (d.getMonth() + 1)).slice(-2);
    const day = ("0" + d.getDate()).slice(-2);
    return `${y}-${m}-${day}`;
  };

  reportData.forEach((row) => {
    const payroll = String(row[0]).trim();
    const name = row[1];
    const dateValue = row[2]; // POPRAWIONE: Data jest w kolumnie C (indeks 2)
    const reqId = String(row[3]).trim(); // POPRAWIONE: Request ID jest w kolumnie D (indeks 3)

    if (!payroll || !dateValue || payroll === "") return;

    // Pobierz Departament z zakładki Users
    const userRow = userValues.find(r => String(r[0]).trim() === payroll);
    const dept = userRow ? userRow[5] : "";

    const d = new Date(dateValue);
    if (isNaN(d.getTime())) return; // Zabezpieczenie przed złą datą

    const dayOfWeek = d.getDay(); 

    const sun = new Date(d);
    sun.setDate(d.getDate() - dayOfWeek);
    sun.setHours(0,0,0,0);
    const sunStr = toDateStr(sun);

    let rowIndex = appValues.findIndex(r => 
      String(r[1]).trim() === payroll && 
      toDateStr(r[4]) === sunStr
    );

    let targetRow;
    if (rowIndex !== -1) {
      targetRow = rowIndex + 1;
      // Jeśli wiersz istnieje, ale nie ma ID, dodajmy to ID
      if (!appSheet.getRange(targetRow, 1).getValue()) {
          appSheet.getRange(targetRow, 1).setValue(reqId);
      }
    } else {
      // Jeśli wiersz nie istnieje, twórzmy nowy (od razu z prawidłowym Request ID!)
      appSheet.appendRow([reqId, payroll, name, dept, sun, "", "", "", "", "", "", "", 0, new Date(), "RECON_FIX"]);
      targetRow = appSheet.getLastRow();
      appValues.push([reqId, payroll, name, dept, sun]); 
    }

    const targetCol = colMap[dayOfWeek];
    appSheet.getRange(targetRow, targetCol).setValue("H");
    
    const weekRow = appSheet.getRange(targetRow, 6, 1, 7).getValues()[0];
    let total = 0;
    weekRow.forEach(val => {
        const cleanVal = String(val).trim().toUpperCase();
        if (cleanVal === 'H') total += 1;
        if (['HD', 'HD1', 'HD2'].includes(cleanVal)) total += 0.5;
    });
    appSheet.getRange(targetRow, 13).setValue(total);

    uniquePayrolls.add(payroll);
    count++;
  });

  if (count > 0) {
    const currentYear = new Date().getFullYear();
    uniquePayrolls.forEach(p => forceRecalculateUserBalance_(p, currentYear));
    uniquePayrolls.forEach(p => forceRecalculateUserBalance_(p, currentYear + 1)); 
    reportSheet.deleteRows(2, count);
    SpreadsheetApp.getUi().alert(`Applied ${count} entries. Column M is now fixed.`);
  }
}

function migrateApprovedIdsToRequestIds_V4() {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  if (!ADMIN_WHITELIST.map(e => e.toLowerCase()).includes(actorEmail)) throw new Error("Security Alert: Unauthorised access attempt blocked.");

  const ss = getSpreadsheet();
  const reqSheet = ss.getSheetByName("Requests");
  const appSheet = ss.getSheetByName("Approved");  
  
  const reqData = reqSheet.getDataRange().getValues();
  const appData = appSheet.getDataRange().getValues();
  const IDX_REQ_ID = 0, IDX_REQ_PAY = 1, IDX_REQ_JSON = 9;
  const IDX_APP_ID = 0, IDX_APP_PAY = 1, IDX_APP_WC = 4;
  const dayCols = [5,6,7,8,9,10,11]; 
  /* -----------------------------------------------------------
     1️⃣  Build a map of every individual day in each Request
  ----------------------------------------------------------- */
  const dateMap = {}; 
  for (let i = 1; i < reqData.length; i++) {
    const payroll = String(reqData[i][IDX_REQ_PAY]);
    const reqId   = String(reqData[i][IDX_REQ_ID]);
    if (!payroll || !reqId) continue;
    let jsonDates = [];
    try {
      if (reqData[i][IDX_REQ_JSON]) jsonDates = JSON.parse(reqData[i][IDX_REQ_JSON]);
    } catch(e) {}
    jsonDates.forEach(ds => {
      const iso = toDateStr(ds);
      dateMap[`${payroll}_${iso}`] = reqId;
    });
  }
  /* -----------------------------------------------------------
     2️⃣  Examine each Approved row and match by any 'H' cell
  ----------------------------------------------------------- */
  const updates = [];
  let linked = 0;
  for (let j = 1; j < appData.length; j++) {
    const payroll = String(appData[j][IDX_APP_PAY]);
    let foundId = appData[j][IDX_APP_ID]; 
    const wc = new Date(appData[j][IDX_APP_WC]);
    if (isNaN(wc)) { updates.push([foundId]); continue; }
    for (let d = 0; d < 7; d++) {
      const val = String(appData[j][dayCols[d]]).trim().toUpperCase();
      if (['H', 'HD', 'HD1', 'HD2'].includes(val)) {
        const day = new Date(wc);
        day.setDate(wc.getDate() + d);
        const key = `${payroll}_${toDateStr(day)}`;
        if (dateMap[key]) {
          foundId = dateMap[key];
          linked++;
          break;
        }
      }
    }
    updates.push([foundId]);
  }
  /* -----------------------------------------------------------
     3️⃣  Write the results back
  ----------------------------------------------------------- */
  appSheet.getRange(2, 1, updates.length, 1).setValues(updates);
  Logger.log(`✓ Migration complete: ${linked} Approved rows linked to matching Request_IDs.`);
}

/**
 * REVERSE SYNC TOOL:
 * Scans the 'Approved' tab for orphaned holidays (missing or invalid IDs).
 * Generates a new ID, creates a matching row in the 'Requests' tab, and links them perfectly.
 */
// =================================================================================================================================================================================================
function reverseSyncMissingRequests() {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  if (!ADMIN_WHITELIST.map(e => e.toLowerCase()).includes(actorEmail)) throw new Error("Security Alert: Unauthorised access attempt blocked.");

  const ss = SpreadsheetApp.openById('1wLTDovDcceXdVEK4H0R-2L4OkrAgW8f61pKYYWIPCA4');
  const appSheet = ss.getSheetByName("Approved");
  const reqSheet = ss.getSheetByName("Requests");

  const appData = appSheet.getDataRange().getValues();
  const reqData = reqSheet.getDataRange().getValues();

  // 1. Build a quick lookup dictionary of all valid Request IDs currently in the system
  const validReqIds = new Set();
  for (let i = 1; i < reqData.length; i++) {
    if (reqData[i][0]) validReqIds.add(String(reqData[i][0]).trim());
  }

  const newRequests = [];
  const idUpdates = []; // For rewriting Column A of the Approved sheet
  let fixedCount = 0;

  // Helper to format dates correctly for JSON
  const toDateStr = (d) => {
    const y = d.getFullYear();
    const m = ("0" + (d.getMonth() + 1)).slice(-2);
    const day = ("0" + d.getDate()).slice(-2);
    return `${y}-${m}-${day}`;
  };

  // 2. Loop through every row in the Approved tracker
  for (let i = 1; i < appData.length; i++) {
    const row = appData[i];
    let currentId = row[0] ? String(row[0]).trim() : "";
    const payroll = row[1];
    const name = row[2];
    const wc = row[4];
    
    // Skip if there is no Week Commencing date
    if (!(wc instanceof Date) || isNaN(wc.getTime())) {
      idUpdates.push([currentId]);
      continue;
    }

    let hasHoliday = false;
    let dayCount = 0;
    const specificDates = [];

    // Check Sunday to Saturday (Columns F through L)
    for (let dIdx = 0; dIdx < 7; dIdx++) {
      const val = String(row[5 + dIdx]).trim().toUpperCase();
      if (['H', 'HD', 'HD1', 'HD2'].includes(val)) {
        hasHoliday = true;
        dayCount += (val === 'H' ? 1 : 0.5);
        
        const d = new Date(wc);
        d.setDate(wc.getDate() + dIdx);
        specificDates.push(toDateStr(d));
      }
    }

    // 3. THE FIX: If there are holidays, BUT the ID is blank or doesn't exist in Requests...
    if (hasHoliday && (!currentId || !validReqIds.has(currentId))) {
      
      // Generate a brand new, unique Reverse-Sync ID
      const newId = `REV-${Date.now()}-${Math.floor(Math.random() * 10000)}`;
      currentId = newId;

      const startStr = specificDates[0];
      const endStr = specificDates[specificDates.length - 1];
      const jsonStr = JSON.stringify(specificDates);

      // Construct the new row for the Requests sheet
      // Format aligns with: [Request_ID, User_ID, Name, Start_Date, End_Date, Days, Status, Manager_Email, Timestamp, Raw_Dates_JSON]
      newRequests.push([
        newId, payroll, name, startStr, endStr, dayCount, 'Approved', 'Reverse Sync Tool', new Date(), jsonStr
      ]);

      validReqIds.add(newId); // Add to set to prevent micro-collisions
      fixedCount++;
    }

    // Push the ID (whether it's the original one or the newly generated one) to our update list
    idUpdates.push([currentId]);
  }

  // 4. Write all the new "dummy" requests into the Requests sheet in one fast batch
  if (newRequests.length > 0) {
    reqSheet.getRange(reqSheet.getLastRow() + 1, 1, newRequests.length, newRequests[0].length).setValues(newRequests);
  }

  // 5. Overwrite Column A of the Approved sheet so the links are made permanent
  if (idUpdates.length > 0) {
    appSheet.getRange(2, 1, idUpdates.length, 1).setValues(idUpdates);
  }

  console.log(`Reverse Sync Complete! Created ${fixedCount} missing requests and linked them perfectly.`);
}

/**
 * FORWARD SYNC TOOL:
 * Scans the 'Requests' tab for Approved holidays.
 * Checks if those specific days exist in the 'Approved' tracker.
 * If missing, it safely injects the 'H' and links the correct Request ID.
 */
// =================================================================================================================================================================================================
function forwardSyncMissingApproved() {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  if (!ADMIN_WHITELIST.map(e => e.toLowerCase()).includes(actorEmail)) throw new Error("Security Alert: Unauthorised access attempt blocked.");

  const ss = SpreadsheetApp.openById('1wLTDovDcceXdVEK4H0R-2L4OkrAgW8f61pKYYWIPCA4');
  const reqSheet = ss.getSheetByName("Requests");
  const appSheet = ss.getSheetByName("Approved");
  const userSheet = ss.getSheetByName("Users");

  const reqData = reqSheet.getDataRange().getValues();
  const appData = appSheet.getDataRange().getValues();
  const userData = userSheet.getDataRange().getValues();

  // Helper to standardise dates
  const toDateStr = (input) => {
    if (!input) return "";
    const d = (input instanceof Date) ? input : new Date(input);
    if (isNaN(d.getTime())) return "";
    const y = d.getFullYear();
    const m = ("0" + (d.getMonth() + 1)).slice(-2);
    const day = ("0" + d.getDate()).slice(-2);
    return `${y}-${m}-${day}`;
  };

  // 1. Build a map of Users for quick lookup (to get Names and Departments)
  const userMap = {};
  for (let i = 1; i < userData.length; i++) {
    userMap[String(userData[i][0]).trim()] = {
      payroll: String(userData[i][0]).trim(),
      name: userData[i][1],
      dept: userData[i][5]
    };
  }

  // 2. Build an index of exactly what is currently in the Approved tracker to avoid overwriting
  const trackerIndex = new Set();
  const dayCols = [5, 6, 7, 8, 9, 10, 11]; // Columns F through L (Sun-Sat)
  
  for (let i = 1; i < appData.length; i++) {
    const payroll = String(appData[i][1]).trim();
    const wc = appData[i][4];
    if (!(wc instanceof Date)) continue;
    
    dayCols.forEach((colIdx, dIdx) => {
      const val = String(appData[i][colIdx]).trim().toUpperCase();
      // If it's H, HD, or even C (Cancelled), we know the system has a record of it
      if (['H', 'HD', 'HD1', 'HD2', 'C'].includes(val)) {
        const d = new Date(wc);
        d.setDate(wc.getDate() + dIdx);
        trackerIndex.add(`${payroll}_${toDateStr(d)}`);
      }
    });
  }

  // 3. Find missing dates and group them to send to the database
  let missingCount = 0;
  const injectionsToProcess = {}; // Grouped by Request ID

  for (let i = 1; i < reqData.length; i++) {
    const reqId = String(reqData[i][0]).trim();
    const payroll = String(reqData[i][1]).trim();
    const daysVal = Number(reqData[i][5]);
    const status = String(reqData[i][6]).trim();
    const jsonDates = reqData[i][9];

    if (status === 'Approved' && jsonDates && userMap[payroll]) {
      try {
        const dates = JSON.parse(jsonDates);
        dates.forEach(dateStr => {
          const isoDate = toDateStr(dateStr);
          
          // If this specific date is NOT in the tracker for this person...
          if (!trackerIndex.has(`${payroll}_${isoDate}`)) {
            if (!injectionsToProcess[reqId]) {
               injectionsToProcess[reqId] = {
                  userObj: userMap[payroll],
                  dates: [],
                  code: (daysVal === 0.5) ? 'HD1' : 'H'
               };
            }
            injectionsToProcess[reqId].dates.push(new Date(dateStr));
            trackerIndex.add(`${payroll}_${isoDate}`); // Prevent duplicate injections
            missingCount++;
          }
        });
      } catch(e) {
        console.warn(`Failed to parse dates for ${reqId}`);
      }
    }
  }

  if (missingCount === 0) {
    console.log("No missing holidays found. The tracker is perfectly up to date!");
    return;
  }

  console.log(`Found ${missingCount} missing days. Injecting them into the Tracker now...`);

  // 4. Inject the missing days using your system's built-in writeToWeeklyDB function
  // This guarantees they are put in the right Week Commencing rows.
  for (const [reqId, data] of Object.entries(injectionsToProcess)) {
      writeToWeeklyDB_(data.userObj, data.dates, data.code, "Forward Sync Tool", reqId);
  }
  
  // 5. Force a recalculation of balances for affected users
  const affectedPayrolls = [...new Set(Object.values(injectionsToProcess).map(d => d.userObj.payroll))];
  const currentYear = new Date().getFullYear();
  affectedPayrolls.forEach(p => {
      // Recalc for current and next year just to be safe
      forceRecalculateUserBalance_(p, currentYear);
      forceRecalculateUserBalance_(p, currentYear + 1);
  });

  console.log(`Forward Sync Complete! Restored ${missingCount} missing days and updated balances.`);
}

        if (role !== 'Admin' && role !== 'AdminManager') {
             return { success: false, error: `Permission Denied. The system thinks your role is: [${role}]` };

function migrateLegacyHalfDays() {
  const actorEmail = Session.getActiveUser().getEmail().toLowerCase();
  if (!ADMIN_WHITELIST.map(e => e.toLowerCase()).includes(actorEmail)) throw new Error("Security Alert: Unauthorised access attempt blocked.");

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const dbSheet = ss.getSheetByName('Approved');
  const data = dbSheet.getDataRange().getValues();
  let updatedCount = 0;
  const dayCols = [5, 6, 7, 8, 9, 10, 11];
  for (let i = 1; i < data.length; i++) {
    let rowUpdated = false;
    dayCols.forEach(colIdx => {
      if (String(data[i][colIdx]).trim().toUpperCase() === 'HD') {
        data[i][colIdx] = 'HD1';
        rowUpdated = true;
      }
    });
    if (rowUpdated) {
      dbSheet.getRange(i + 1, 6, 1, 7).setValues([data[i].slice(5, 12)]);
      updatedCount++;
    }
  }
  console.log(`Migration Complete. Updated ${updatedCount} rows containing legacy HD codes to HD1.`);
}

// =======================================================================================
// NIGHTLY MAINTENANCE ENGINE
// Syncs Departments from Users -> Approved & Clears Cancelled Christmas Reservations
// =======================================================================================
function runNightlyMaintenance() {
  console.time("Nightly Maintenance");
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const userSheet = ss.getSheetByName(SHEET_NAMES.USERS);
  const appSheet = ss.getSheetByName(SHEET_NAMES.DB_WEEKLY);
  
  const userData = userSheet.getDataRange().getValues();
  const appData = appSheet.getDataRange().getValues();
  
  const currentYear = new Date().getFullYear();
  
  // 1. Build User Map (Source of Truth)
  const userDeptMap = {};
  userData.forEach((u, i) => {
    if (i === 0) return;
    userDeptMap[String(u[0]).trim()] = String(u[5]).trim();
  });
  
  // 2. Process Approved Tab (Dept Sync & Xmas Scan)
  let appUpdated = false;
  const xmasBookedPayrolls = new Set();
  
  for (let i = 1; i < appData.length; i++) {
    const row = appData[i];
    const payroll = String(row[1]).trim();
    const wc = row[4];
    
    // A. Department Sync
    if (userDeptMap[payroll] && row[3] !== userDeptMap[payroll]) {
      appData[i][3] = userDeptMap[payroll];
      appUpdated = true;
    }
    
    // B. Scan for Active Christmas Bookings (Dec 20 - Jan 4)
    if (wc instanceof Date) {
        for (let dIdx = 0; dIdx < 7; dIdx++) {
            const code = String(row[5 + dIdx]).toUpperCase();
            if (['H', 'HD', 'HD1', 'HD2'].includes(code)) {
                const d = new Date(wc);
                d.setDate(d.getDate() + dIdx);
                const m = d.getMonth();
                const day = d.getDate();
                
                // If the active holiday falls in the festive window for the current rota year
                if ((m === 11 && day >= 20 && d.getFullYear() === currentYear) || 
                    (m === 0 && day <= 4 && d.getFullYear() === currentYear + 1)) {
                    xmasBookedPayrolls.add(payroll);
                }
            }
        }
    }
  }
  
  if (appUpdated) {
    appSheet.getRange(1, 1, appData.length, appData[0].length).setValues(appData);
    console.log("Departments synced.");
  }
  
  // 3. Process Users Tab (Christmas Cleanup)
  let usersUpdated = false;
  for (let i = 1; i < userData.length; i++) {
    const payroll = String(userData[i][0]).trim();
    const lastOff = Number(userData[i][15]); // Column P
    
    // If they are marked as having this Christmas off, but no active booking was found
    if (lastOff === currentYear && !xmasBookedPayrolls.has(payroll)) {
        userData[i][15] = ""; // Clear the year
        usersUpdated = true;
    }
  }
  
  if (usersUpdated) {
    userSheet.getRange(1, 1, userData.length, userData[0].length).setValues(userData);
    console.log("Cleared cancelled Christmas reservations.");
  }
  
  console.timeEnd("Nightly Maintenance");
}

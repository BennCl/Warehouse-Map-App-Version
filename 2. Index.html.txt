    // ===============================================================================================================================================================================
    function calculateShortBreaksUnified(requests, patternCode, viewYear) {
        // 1. Gather all valid dates
        let allDates = [];
        
        requests.forEach(r => {
            // Filter: Approved only
            if (r.status !== 'Approved') return;
            
            // Filter: Ignore Statutory / Company Benefit / Admin non-holidays
            if (r.source === 'Statutory' || r.source === 'Company Benefit') return;
            if (r.name === "New Year's Day" || r.name === "Christmas Day" || r.name === "Boxing Day") return;
            
            // STRICT ISOLATION: Ignore all Half Days from Admin Logs and Requests
            if (r.source === 'Admin Log' && r.code !== 'H') return;
            if (Number(r.days) === 0.5 || ['HD', 'HD1', 'HD2'].includes(r.code)) return;

            // Get Dates
            let dates = r.specificDates || [];
            if (dates.length === 0) { // Legacy fallback
                let d = new Date(r.start);
                const end = new Date(r.end);
                while(d <= end) { dates.push(getSafeLocalISO(d)); d.setDate(d.getDate()+1); }
            }
            
            dates.forEach(ds => {
                const dObj = new Date(ds);
                if (dObj.getFullYear() === viewYear) {
                    dObj.setHours(12,0,0,0); // Force Noon to avoid DST issues
                    allDates.push(dObj.getTime());
                }
            });
        });

        // 2. Sort & Deduplicate
        const uniqueSorted = [...new Set(allDates)].sort((a,b) => a - b);

        // 3. Calculate with Gap Bridge
        let sbCount = 0;
        let currentRun = 0;

        for (let i = 0; i < uniqueSorted.length; i++) {
            currentRun++;
            const current = new Date(uniqueSorted[i]);
            const next = uniqueSorted[i+1] ? new Date(uniqueSorted[i+1]) : null;

            let isConnected = false;
            if (next) {
                const diffTime = next - current;
                const diffDays = Math.round(diffTime / (1000 * 60 * 60 * 24));

                if (diffDays === 1) {
                    isConnected = true; // Consecutive days
                } else {
                    // --- GAP CHECK ---
                    // If the gap consists ONLY of Rest Days, we bridge it.
                    let gapIsClean = true;
                    let temp = new Date(current);
                    temp.setDate(temp.getDate() + 1);
                    
                    // Walk the gap day by day
                    while (temp < next) {
                        // Check using the specific pattern code for this user
                        if (isWorkingDay(temp, patternCode)) { 
                            gapIsClean = false; // Found a working day in gap -> Chain Broken
                            break; 
                        }
                        temp.setDate(temp.getDate() + 1);
                    }
                    if (gapIsClean) isConnected = true;
                }
            }

            if (!isConnected) {
                // End of block. If run was <= 2 days, it's a short break.
                if (currentRun <= 2) sbCount++;
                currentRun = 0;
            }
        }
        return sbCount;
    }

        let filtered = STATE.teamMembers.filter(m => {

            const deptMatch = SELECTED_TEAM_DEPTS.includes('All') || SELECTED_TEAM_DEPTS.includes(m.department);
            let basicMatch = nameMatch && deptMatch;

            // 2. Christmas Filter Override
            if (isXmasFilterActive) {
                if (!basicMatch) return false; 

                // FIX: Check BOTH years (CY and NY)

                const resCY = m.xmas ? Number(m.xmas.reserveCY || 0) : 0;
                const resNY = m.xmas ? Number(m.xmas.reserveNY || 0) : 0;
                const lastOff = m.xmas ? Number(m.xmas.lastOff || 0) : 0;

                // Condition 1: Has a reserve in either year (Red Badge / Action Needed)
                const hasAction = (resCY > 0 || resNY > 0);

                // Condition 2: Is confirmed booked for either year (Green Badge)
                const isBooked = (lastOff === currentYear || lastOff === nextYear);

                return hasAction || isBooked;
            }

            return basicMatch;
        });

        // Sort Logic
        filtered.sort((a, b) => {
            let valA, valB;
            if (currentSort === 'Name') { valA = a.name; valB = b.name; }
            else if (currentSort === 'Allowance') { valA = a.allowance.cy.remaining; valB = b.allowance.cy.remaining; }
            else if (currentSort === 'Short') { valA = a.shortBreaks; valB = b.shortBreaks; }
            
            // --- NEW HEADER SORTS ---
            else if (currentSort === 'LastOff') { 
                valA = Number(a.xmas.lastOff || 0); 
                valB = Number(b.xmas.lastOff || 0); 
            }
            else if (currentSort === 'XmasCY') { 
                // Prioritize "Refused" (-1) -> "Action" (>0) -> "Booked" (0/null)
                const score = (v) => v === 'Refused' ? -1 : Number(v || 0);
                valA = score(a.xmas.reserveCY); valB = score(b.xmas.reserveCY); 
            }
            else if (currentSort === 'XmasNY') { 
                const score = (v) => v === 'Refused' ? -1 : Number(v || 0);
                valA = score(a.xmas.reserveNY); valB = score(b.xmas.reserveNY); 
            }
            
            if (valA < valB) return sortAsc ? -1 : 1; 
            if (valA > valB) return sortAsc ? 1 : -1; 
            return 0;
        });
        
        if (currentSort === 'Short' && sortAsc === true) filtered.reverse();
        
        renderTeamTable(filtered);
    }

    function getBookingData(date) {
        if (!bookingTarget || !bookingTarget.bookedRanges) return null;
        
        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, '0');
        const day = String(date.getDate()).padStart(2, '0');
        const localISO = `${year}-${month}-${day}`;

        return bookingTarget.bookedRanges.find(range => {
                // 1. Check specific dates
                if (range.specificDates && range.specificDates.length > 0) {
                    return range.specificDates.includes(localISO);
                }
                // 2. Check range (Legacy support)
                return localISO >= range.from && localISO <= range.to;
        }) || null;
    }

    // --- OTHER ACTIONS ---
    function openAmendModal(reqId) {
    currentAmendId = reqId;
    
    // 1. Find ALL fragments of this request across the dashboard
    const fragments = STATE.requests.filter(r => r.id === reqId).concat(STATE.teamRequests.filter(r => r.id === reqId));
    
    if(fragments.length === 0) {
        showCustomAlert("Error: Request data not found.", "error");
        return;
    }

    const list = document.getElementById('amendDatesList'); 
    list.innerHTML = '';
    
    let allDates = [];
    
    // 2. Combine dates from all fragments into one array
    fragments.forEach(req => {
        let dates = req.specificDates || [];
        
        if(dates.length === 0) { 
            let d = new Date(req.start); 
            const end = new Date(req.end);
            while(d <= end) { 
                dates.push(getSafeLocalISO(d)); 
                d.setDate(d.getDate()+1); 
            } 
        }
        allDates = allDates.concat(dates);
    });
    
    // 3. Deduplicate and sort the dates chronologically
    const uniqueDates = [...new Set(allDates)].sort();
    
    // 4. Set up the Date rules for locking past days & 7-day rule
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const startOfWeek = new Date(today);
    startOfWeek.setDate(today.getDate() - today.getDay());
    
    // 5. Build the checkbox list with polished UI locks
    uniqueDates.forEach(dateStr => { 
        const dObj = new Date(dateStr);
        dObj.setHours(0, 0, 0, 0);
        let isLocked = false;
        let lockBadge = ""; 
        
        const diffTime = dObj.getTime() - today.getTime();
        const diffDays = Math.round(diffTime / (1000 * 60 * 60 * 24));
        
        // Managers/Admins can edit the current week
        if (STATE.currentUser.rank >= 30) { 
            if (dObj < startOfWeek) {
                isLocked = true;
                lockBadge = "<i class='fas fa-lock'></i> Past Date";
            }
        } else {
            // Regular users cannot edit past/today
            if (dObj <= today) {
                isLocked = true;
                lockBadge = "<i class='fas fa-lock'></i> Past Date";
            } 
            // C3 Colleagues locked out if within 7 days
            else if (STATE.currentUser.accessCode === 'C3' && diffDays < 7) {
                isLocked = true;
                lockBadge = "<i class='fas fa-lock'></i> 7-Day Rule";
            }
        }

        const li = document.createElement('li'); 
        li.className = 'amend-item'; 
        
        if (isLocked) {
            // Polished Flexbox Label with a proper Status Badge
            li.innerHTML = `<input type="checkbox" id="d-${dateStr}" value="${dateStr}" checked disabled style="opacity: 0.5; cursor:not-allowed;">

    function executeCancellation() {
        if(!pendingCancelId) return;
        const btn = document.getElementById('confirmActionBtn');
        const txt = btn.textContent;
        btn.textContent = "Processing..."; btn.disabled = true;
        
        let datesToCancel = [];
        // Find the request object in our local state
        const req = STATE.teamRequests.find(r => r.id === pendingCancelId) || STATE.requests.find(r => r.id === pendingCancelId);
        
        if (req) {
            // FIX: Check for specific dates first! 
            // This prevents overwriting gaps in split bookings.
            if (req.specificDates && req.specificDates.length > 0) {
                // Use the exact dates we already know about
                // Ensure they are strings (ISO format)
                datesToCancel = req.specificDates.map(d => {
                    // Handle case where specificDates might be Date objects or strings
                    if (d instanceof Date) return getSafeLocalISO(d);
                    return d.substring(0, 10); // Safe ISO string
                });
            } else {
                // Fallback: Loop from Start to End (Legacy behavior)
                let d = new Date(req.start);
                const end = new Date(req.end);
                while (d <= end) { 
                    datesToCancel.push(getSafeLocalISO(d)); 
                    d.setDate(d.getDate() + 1);
                }
            }
        }

        google.script.run.withSuccessHandler(res => {
            btn.textContent = txt; btn.disabled = false; closeBookingModal('confirmModal'); closeBookingModal('viewColleagueModal');
            if(res.success) { 
                showCustomAlert(res.message, "success"); 
                google.script.run.withSuccessHandler(initApp).getInitialData();
                if(STATE.config.showTeamDashboard) loadManagerDashboard();
            } else { showCustomAlert(res.error, "error"); }
        }).attemptCancellation({ reqId: pendingCancelId, datesToCancel: datesToCancel });
    }

    function isWorkingDay(date, patternOverride) {
    const m = date.getMonth();
    const day = date.getDate();
    const year = date.getFullYear();

    const isFestiveWindow = (m === 11 && day >= 20) || (m === 0 && day <= 4);

    if (isFestiveWindow && STATE.xmasExceptions && STATE.statutory) {
        let rotaYear = year;
        if (m === 0 && day <= 4) rotaYear = year - 1;

        let xmasGroup = "";
        if (bookingTarget && bookingTarget.xmasGroup) {
            xmasGroup = bookingTarget.xmasGroup;
        } else if (!bookingTarget && STATE.currentUser && STATE.currentUser.xmasGroup) {
            xmasGroup = STATE.currentUser.xmasGroup;
        }

        if (!xmasGroup && STATE.statutory) {
            const dept = bookingTarget ? bookingTarget.department : STATE.currentUser.department;
            const cleanDept = String(dept || "").trim().toLowerCase();
            const keysToTry = [`${cleanDept}_${rotaYear}`, `${cleanDept}_${year}`, `${cleanDept}_${new Date().getFullYear()}`];
            for (let k of keysToTry) {
                if (STATE.statutory[k] && STATE.statutory[k].xmasGroup) {
                    xmasGroup = STATE.statutory[k].xmasGroup;
                    break;
                }
            }
        }

        if (xmasGroup) {
            const iso = getSafeLocalISO(date);
            let exceptionResult = undefined;
            const strictKey = `${rotaYear}_${xmasGroup}`;

            if (STATE.xmasExceptions[strictKey] && STATE.xmasExceptions[strictKey][iso] !== undefined) {
                exceptionResult = STATE.xmasExceptions[strictKey][iso];
            } else {
                for (let key in STATE.xmasExceptions) {
                    if (key.endsWith(`_${xmasGroup}`) && STATE.xmasExceptions[key][iso] !== undefined) {
                        exceptionResult = STATE.xmasExceptions[key][iso];
                        break;
                    }
                }
            }
            if (exceptionResult !== undefined) return exceptionResult;
        }
    }

    const patternCode = patternOverride || (bookingTarget ? bookingTarget.pattern : STATE.currentUser.workPattern);
    if(!STATE.patterns || !STATE.patterns[patternCode]) { 
        const d = date.getDay();
        return d !== 0 && d !== 6;
    } 

    const d = new Date(date);
    const utcDate = Date.UTC(d.getFullYear(), d.getMonth(), d.getDate());
    const utcEpoch = Date.UTC(2024, 11, 29);
    const diffMs = utcDate - utcEpoch;
    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
    
    const cycleLength = STATE.patterns[patternCode].cycleLength;
    const weekIndex = Math.floor(diffDays / 7) % cycleLength;
    const normalizedIndex = weekIndex >= 0 ? weekIndex : weekIndex + cycleLength;
    const dayName = d.toLocaleDateString('en-GB', { weekday: 'short' });
    const weekDays = STATE.patterns[patternCode].weeks[normalizedIndex];
    
    return weekDays && weekDays.includes(dayName);
}

    function calculateProRata() {
        const type = document.getElementById('calcType').value;
        const dateStr = document.getElementById('calcDate').value;
        const full = parseFloat(document.getElementById('calcFullAllow').value);
        
        if(!dateStr || !full) return showCustomAlert("Please enter a valid date and allowance.", "error");

        const inputDate = new Date(dateStr);
        const startOfYear = new Date(inputDate.getFullYear(), 0, 1);
        const endOfYear = new Date(inputDate.getFullYear(), 11, 31);
        const daysInYear = (endOfYear - startOfYear) / 86400000 + 1;
        
        let relevantDays = (type === 'Starter') ?
            (endOfYear - inputDate) / 86400000 + 1 : (inputDate - startOfYear) / 86400000 + 1;
        
        if(relevantDays < 0) relevantDays = 0;
        
        // UPDATED: Use Math.round() for standard rounding (0.5 rounds up)
        const rawResult = (relevantDays / daysInYear) * full;
        const finalResult = Math.round(rawResult);
        
        document.getElementById('calcResult').textContent = finalResult + " Days";
    }

    function calculateDynamicProRata() {
        const fullAllowance = parseFloat(document.getElementById('splitFullAllow').value) || 0;
        
        if (fullAllowance === 0) {
            showCustomAlert("Please enter the Standard Full Time Allowance.", "error");
            return;
        }

        const rows = document.querySelectorAll('.period-row');
        let totalEntitlement = 0;
        let totalWeeks = 0;

        rows.forEach(row => {
            const w = parseFloat(row.querySelector('.p-weeks').value) || 0;
            const d = parseFloat(row.querySelector('.p-days').value) || 0;
            
            // Formula: (FullAllow * (DaysWorked / 5)) * (WeeksWorked / 52)
            const periodEntitlement = (fullAllowance * (d / 5)) * (w / 52);
            
            totalEntitlement += periodEntitlement;
            totalWeeks += w;
        });

        // UPDATED: Use Math.round() for standard rounding (0.5 rounds up)
        const finalResult = Math.round(totalEntitlement);

        document.getElementById('splitResult').textContent = finalResult + " Days";
        updatePeriodTotalsDisplay(); 
        
        if (totalWeeks > 52) {
             showCustomAlert("Warning: Total weeks entered (" + totalWeeks + ") exceeds 52.", "error");
        }
    }

    function populateUserFormDropdowns() {
    // 1. Populate Patterns (Instant - from local STATE)
    const patSel = document.getElementById('uPattern');
    
    // --- HELPER: Compress "Sun, Mon, Tue" to "Sun-Tue" ---
    const compressDays = (str) => {
        if (!str) return "";
        const map = { "Sun":0, "Mon":1, "Tue":2, "Wed":3, "Thu":4, "Fri":5, "Sat":6 };
        const rMap = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
        let indices = str.split(',')
            .map(s => s.trim())
            .filter(s => map.hasOwnProperty(s))
            .map(s => map[s])
            .sort((a,b) => a-b);
        if (indices.length === 0) return str;
        let ranges = [];
        let start = indices[0];
        let prev = indices[0];
        for (let i = 1; i < indices.length; i++) {
            if (indices[i] === prev + 1) {
                prev = indices[i];
            } else {
                if (prev - start >= 2) ranges.push(`${rMap[start]}-${rMap[prev]}`);
                else if (prev - start === 1) ranges.push(`${rMap[start]}, ${rMap[prev]}`);
                else ranges.push(rMap[start]);
                start = indices[i];
                prev = indices[i];
            }
        }
        if (prev - start >= 2) ranges.push(`${rMap[start]}-${rMap[prev]}`);
        else if (prev - start === 1) ranges.push(`${rMap[start]}, ${rMap[prev]}`);
        else ranges.push(rMap[start]);
        return ranges.join(", ");
    };

    if (patSel.options.length <= 1 && STATE.patterns) {
        patSel.innerHTML = '<option value="">Select Pattern...</option>';
        Object.keys(STATE.patterns).sort().forEach(p => {
            const data = STATE.patterns[p];

            if (data && data.weeks && data.weeks.length > 0) {
                const validWeeks = data.weeks.filter(w => w && w.toString().trim() !== "");

                    const readableWeeks = validWeeks.map(w => compressDays(w.toString()));

    const deptSel = document.getElementById('uDept');
    if (deptSel.options.length <= 1) {
        deptSel.innerHTML = '<option value="">Select Department...</option>';
        
        let depts = [];
        
        // Strategy A: Get every unique department currently in the database 
        if (STATE.teamMembers && STATE.teamMembers.length > 0) {
            depts = [...new Set(STATE.teamMembers.map(m => m.department))];
        } 
        
        // Strategy B: Merge with your comprehensive list to ensure even departments with no users yet are visible
        const masterList = [
            "Warehouse AM", "Warehouse PM", "Warehouse NT", "Clerical AM", "Clerical PM", "Clerical NT",
            "QC AM", "QC PM", "QC/PI", "PI AM", "PI PM 1", "PI PM 2", "PI NT",
            "Stop & Check AM", "Stop & Check PM", "Stop & Check NT",
            "Trainer AM", "Trainer PM", "Trainer NT", "MHE Trainer",
            "Facilities AM", "Facilities Ops Manager", "Facilities Supervisor",
            "NLL AM", "CI/NLL", "Continuous Improvement", "Compliance Analyst",
            "Admin", "AGM", "General Manager", "DIDOS AM", "DIDOS LM",
            "Supervisor AM", "Supervisor PM", "Supervisor NT",
            "Line Manager AM", "Line Manager PM", "Line Manager NT",
            "Health and Safety Line Manager", "Transport Development Manager", "Transport Fleet Manager",
            "Ops Manager AM", "Ops Manager PM", "Ops Manager NT", "Transport Ops Manager",
            "MCFC OPS Manager AM", "MCFC AM", "MCFC PM", "MCFC NT", 
            "MCFC Clerical AM", "MCFC Clerical PM", "MCFC PI AM",
            "MCFC Supervisor AM", "MCFC Supervisor PM", "MCFC Supervisor NT",
            "MCFC Line Manager AM", "MCFC Line Manager PM", "MCFC Line Manager NT",

    let sessionTimeout;
    let sessionListenersAttached = false; // New flag to prevent duplicate listeners

    function startSessionTimer() {
        // 1. CLEAR EXISTING TIMER (Fixes the Ghost Bug)
        if (sessionTimeout) clearTimeout(sessionTimeout);

        // 2. Determine User Role
        const role = STATE.currentUser ? STATE.currentUser.accessCode : 'C3'; 
        const LONG_SESSION_ROLES = ['C6', 'M1', 'M2', 'M3', 'Admin', 'AdminManager'];
        const limit = LONG_SESSION_ROLES.includes(role) ? (10 * 60 * 1000) : (3 * 60 * 1000);
        
        console.log(`Session Timer Set: ${limit / 60000} minutes for ${role}`);

        // 3. Attach Listeners (ONLY ONCE)
        if (!sessionListenersAttached) {
            const activityEvents = ['mousemove', 'keypress', 'click', 'scroll', 'touchstart'];
            activityEvents.forEach(evt => {
                document.addEventListener(evt, () => {
                    if (sessionTimeout) clearTimeout(sessionTimeout);
                    sessionTimeout = setTimeout(forceHardLogout, limit);
                });
            });
            sessionListenersAttached = true; 
        }

        // 4. Start/Restart the Timer
        sessionTimeout = setTimeout(forceHardLogout, limit);
    }

                            <td><em>Full Time Equivalent</em></td><td class="breakdown-val" style="color:#888;"><em>${d.fteTotal}</em></td>
                         </tr>
                         <tr class="breakdown-row" style="background:#fffcf0; color:#888; font-size:0.9em;">
                            <td><em>Pro-Rata (${d.daysPerWeek} days)</em></td><td class="breakdown-val" style="color:#888;"><em>${d.proRataEntitlement}</em></td>
                         </tr>`;
            }

            // BOUGHT DAYS (M1+ Visibility Check)
            // M1 is Rank 40.
            if (STATE.currentUser.rank >= 40) {
                 html += `<tr class="breakdown-row"><td>Extra Days Bought</td><td class="breakdown-val">${d.bought}</td></tr>`;
            }

            if (d.adjusted !== 0) {
                const sign = d.adjusted > 0 ? "+" : "";
                html += `<tr class="breakdown-row"><td>Adjustments</td><td class="breakdown-val">${sign}${d.adjusted}</td></tr>`;
            }

            // FINAL TOTAL
            // FIX: Removed the duplicate addition of d.bought
            const displayTotal = (isPartTime ? d.proRataEntitlement : d.fteTotal) 
                                 + d.bought 
                                 + d.adjusted;

            html += `<tr class="breakdown-row" style="font-size:1.1em; border-top:2px solid #ddd;">
                        <td>Total Allowance</td><td class="breakdown-val">${Math.round(displayTotal)}</td>
                     </tr>`;
            tbody.innerHTML = html;

        }).getMyAllowanceBreakdown();
    }

    function submitTransportLog() {
        const payroll = document.getElementById('transportDriverSelect').value;
        const type = document.querySelector('input[name="transType"]:checked').value;
        
        if (!payroll || !transPicker || transPicker.selectedDates.length === 0) {
            showCustomAlert("Please select a driver and at least one date.", "error");
            return;
        }
        
        const dates = transPicker.selectedDates.map(d => getSafeLocalISO(d));

        const btn = document.querySelector('#transportModal .btn-primary');
        const oldText = btn.textContent;
        btn.textContent = "Logging...";
        btn.disabled = true;

        google.script.run.withSuccessHandler(res => {
            btn.textContent = oldText;
            btn.disabled = false;
            document.getElementById('transportModal').style.display = 'none';
            
            if (res.success) {
                showCustomAlert(res.message, "success");
                loadManagerDashboard(); // Refresh the table behind the modal
            } else {
                showCustomAlert(res.error, "error");
            }
        }).transportLogEvent({ targetPayroll: payroll, dates: dates, code: type });
    }

function triggerEngineUpdate() {
        const btn = document.getElementById('btnRunEngine');
        const originalText = btn.innerHTML;
        
        // Lock the UI so they don't spam-click it
        btn.disabled = true;
        btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Running Engine... (This may take a minute)';
        btn.style.backgroundColor = '#555'; // Grey it out
        btn.style.borderColor = '#555';

        google.script.run
            .withSuccessHandler(res => {
                // Reset the button
                btn.disabled = false;
                btn.innerHTML = originalText;
                btn.style.backgroundColor = 'var(--wickes-red)';
                btn.style.borderColor = 'var(--wickes-red)';

                if(res.success) {
                    showCustomAlert(res.message, "success");
                    // Refresh the audit table and dropdowns to reflect new data
                    loadAdminDropdowns();
                    loadAuditHistory();
                } else {
                    showCustomAlert(res.error, "error");
                }
            })
            .withFailureHandler(err => {
                // Reset on failure
                btn.disabled = false;
                btn.innerHTML = originalText;
                btn.style.backgroundColor = 'var(--wickes-red)';
                showCustomAlert("Connection Error: " + err.message, "error");
            })
            .adminTriggerAllowanceEngine();
    }

    function addTeamCheckbox(container, value, label) {
    const div = document.createElement('div');
    div.className = 'multiselect-item';
    // Use replace to strip spaces for valid HTML IDs
    const safeId = `team_chk_${value.replace(/\s+/g, '_')}`;
    div.innerHTML = `<input type="checkbox" value="${value}" id="${safeId}"><label for="${safeId}">${label}</label>`;
    
    div.addEventListener('click', function(e) {
        if (e.target.tagName !== 'INPUT') {
            const chk = div.querySelector('input');
            chk.checked = !chk.checked;
        }
        const target = div.querySelector('input');
        if (value === 'All' && target.checked) {
            container.querySelectorAll('input').forEach(i => { if(i.value !== 'All') i.checked = false; });
        } else if (value !== 'All' && target.checked) {
            const allChk = document.getElementById('team_chk_All');
            if(allChk) allChk.checked = false;
        }
    });
    container.appendChild(div);
}

function applyTeamFilter() {
    const checked = Array.from(document.querySelectorAll('#teamDeptFilterContent .multiselect-item input:checked')).map(c => c.value);
    if (checked.length === 0) {
        SELECTED_TEAM_DEPTS = ["All"];
        document.getElementById('team_chk_All').checked = true;
    } else {
        SELECTED_TEAM_DEPTS = checked;
    }
    
    localStorage.setItem('wickes_team_dept_filter', JSON.stringify(SELECTED_TEAM_DEPTS));
    updateTeamBtnLabel();
    document.getElementById('teamDeptFilterContent').classList.remove('show');
    filterTeamTable(); 
}

function updateTeamBtnLabel() {
    const btn = document.getElementById('teamDeptFilterBtn');
    if (SELECTED_TEAM_DEPTS.includes("All")) {
        btn.textContent = "All Departments";
    } else if (SELECTED_TEAM_DEPTS.length === 1) {
        btn.textContent = SELECTED_TEAM_DEPTS[0];
    } else {
        btn.textContent = `Filter: ${SELECTED_TEAM_DEPTS.length} Selected`;
    }
}

function syncTeamCheckboxes() {
    const boxes = document.querySelectorAll('#teamDeptFilterContent .multiselect-item input');
    boxes.forEach(b => b.checked = false);
    SELECTED_TEAM_DEPTS.forEach(val => {
        const safeId = `team_chk_${val.replace(/\s+/g, '_')}`;
        const el = document.getElementById(safeId);
        if (el) el.checked = true;
    });
}

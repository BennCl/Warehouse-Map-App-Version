<script>
(function() {
  // 1. Define the possible stops
  const STOPS = [
    { 
      id: 'dashboard', 
      sel: '#nav-dashboard a', 
      title: 'My Dashboard', 
      text: 'Your personal hub. View your remaining allowance, booked holidays, and a calendar overview of your year.' 
    },
    {
      id: 'allowance',
      sel: '#remainingDaysDisplay', 
      title: 'Accrued Allowance',
      text: 'Your allowance accrues over the year (20% by Feb, 40% Apr, 60% Jun, 80% Aug, 100% Oct). Ensure you have accrued enough days for the date of your holiday.'
    },
    {
      id: 'short_breaks',
      sel: '#shortBreaksCard', // This will automatically skip if the widget is hidden (for non-C3s)
      title: 'Short Breaks (C3)',
      text: 'You are limited to 5 isolated "Short Breaks" (1-2 day holidays) per year. Longer holidays (3+ days) do not count towards this limit.'
    },
    { 
      id: 'request', 
      sel: '#bookNewHolidayBtnSidebar', 
      title: 'Make a Request', 
      text: 'Click here to book holiday. <br><br><strong>Note:</strong> You cannot self-book dates less than 7 days in advance. If you need short-notice leave, please ask your Line Manager.' 
    },
    {
      id: 'cancellation',
      sel: '#myHolidaysTable',
      title: 'Cancelling Holidays',
      text: 'You can cancel future bookings here. <br><br><strong>Warning:</strong> If you cancel a day, you may not be able to rebook it yourself (e.g. if it is within 7 days). You will need your Manager to rebook it.'
    },
    {
      id: 'statutory',
      sel: '#myHolidaysTable', 
      title: 'Statutory Holidays',
      text: 'You will see Statutory Holidays (New Year, Christmas, Boxing Day) and Paid Absence (Xmas Eve) booked automatically.'
    },
    { 
      id: 'manager', 
      sel: '#nav-manager a', 
      title: 'Team Management', 
      text: 'Managers Only: View your team\'s requests, book holiday on their behalf, and monitor team allowance.' 
    },
    { 
      id: 'tracker', 
      sel: '#nav-tracker a', 
      title: 'Holiday Tracker', 
      text: 'Supervisors & Managers: Access the master tracker for a detailed grid view of all shifts.' 
    },
    { 
      id: 'admin', 
      sel: '#nav-admin a', 
      title: 'Admin Hub', 
      text: 'Admins Only: Manage user accounts, log sickness/absence, and adjust allowances manually.' 
    }
  ];

  let currentSteps = [];
  let stepIndex = 0;

  const els = {
    overlay: document.getElementById('helpTourOverlay'),
    bubble: document.getElementById('helpTourBubble'),
    title: document.getElementById('htTitle'),
    body: document.getElementById('htBody'),
    nextBtn: document.getElementById('htNext'),
    closeBtn: document.getElementById('htClose')
  };

  // 2. Global Function to Inject Button (Called by Index.html)
  window.injectHelpButton = function() {
    if (document.getElementById('helpTourBtn')) return;
    const nav = document.querySelector('.nav-links');
    if (!nav) return;

    const li = document.createElement('li');
    li.style.marginTop = "auto"; // Push to bottom
    li.style.borderTop = "1px solid rgba(255,255,255,0.1)";
    li.innerHTML = `
      <a href="#" id="helpTourBtn">
        <i class="fas fa-question-circle"></i> Help Tour
      </a>`;
    nav.appendChild(li);
    document.getElementById('helpTourBtn').addEventListener('click', (e) => {
      e.preventDefault();
      startTour();
    });
  };

  function startTour() {
    // Build steps dynamically based on what is VISIBLE
    currentSteps = STOPS.filter(stop => {
      const el = document.querySelector(stop.sel);
      if (!el) return false;
      
      // 1. Check Sidebar Items
      const li = el.closest('li');
      if (li) return getComputedStyle(li).display !== 'none';

      // 2. Check Main Content Items (like the table or cards)
      // We check offsetParent to see if the element itself (or its parent) is visible
      return el.offsetParent !== null; 
    });

    if (currentSteps.length === 0) return alert("No tour steps available.");

    stepIndex = 0;
    showStep(0);
  }

  function showStep(idx) {
    if (idx >= currentSteps.length) { endTour(); return; }
    
    const step = currentSteps[idx];
    const target = document.querySelector(step.sel);
    if (!target) { showStep(idx + 1); return; } // Skip if missing

    // Highlight
    document.querySelectorAll('.tour-highlight').forEach(el => el.classList.remove('tour-highlight'));
    
    // If it's a sidebar item, highlight the LI. If it's a table/div, highlight the Element.
    const li = target.closest('li');
    if (li) li.classList.add('tour-highlight');
    else target.classList.add('tour-highlight');

    // UI Updates
    els.overlay.classList.add('show');
    els.bubble.classList.add('show');
    els.title.textContent = step.title;
    els.body.innerHTML = step.text; // Changed to innerHTML to support <br> tags
    els.nextBtn.textContent = (idx === currentSteps.length - 1) ? "Finish" : "Next";

    // Position Bubble (Align with element)
    const rect = target.getBoundingClientRect();
    
    // If sidebar item, place to the right. 
    if (li) {
        els.bubble.style.left = '270px'; // Fixed Sidebar Width
        els.bubble.style.top = Math.max(20, rect.top) + 'px';
    } else {
        // Main content items: Place centered-ish relative to the target
        els.bubble.style.left = Math.max(20, rect.left + (rect.width/2) - 150) + 'px';
        els.bubble.style.top = Math.max(20, rect.top - 10) + 'px'; // Just above/on top
    }
  }

  function nextStep() {
    stepIndex++;
    showStep(stepIndex);
  }

  function endTour() {
    els.overlay.classList.remove('show');
    els.bubble.classList.remove('show');
    document.querySelectorAll('.tour-highlight').forEach(el => el.classList.remove('tour-highlight'));
  }

  // Bind Events
  els.nextBtn.addEventListener('click', nextStep);
  els.closeBtn.addEventListener('click', endTour);
  els.overlay.addEventListener('click', endTour);

})();
</script>

            }
        });

    function isWorkingDay(date, patternCode, dept, explicitXGroup) {
    const m = date.getMonth();
    const day = date.getDate();
    const year = date.getFullYear();

    const isFestiveWindow = (m === 11 && day >= 20) || (m === 0 && day <= 4);

    if (isFestiveWindow && DATA.exceptions) {
        let rotaYear = year;
        if (m === 0 && day <= 4) rotaYear = year - 1;

        let xmasGroup = explicitXGroup || "";

        if (!xmasGroup && DATA.statutory) {
            const cleanDept = String(dept || "").trim().toLowerCase();
            const keysToTry = [`${cleanDept}_${rotaYear}`, `${cleanDept}_${year}`, `${cleanDept}_${new Date().getFullYear()}`];
            for (let k of keysToTry) {
                if (DATA.statutory[k] && DATA.statutory[k].xmasGroup) {
                    xmasGroup = DATA.statutory[k].xmasGroup;
                    break;
                }
            }
        }

        if (xmasGroup) {
            const iso = getSafeLocalISO(date);
            let exceptionResult = undefined;
            const strictKey = `${rotaYear}_${xmasGroup}`;

            if (DATA.exceptions[strictKey] && DATA.exceptions[strictKey][iso] !== undefined) {
                exceptionResult = DATA.exceptions[strictKey][iso];
            } else {
                for (let key in DATA.exceptions) {
                    if (key.endsWith(`_${xmasGroup}`) && DATA.exceptions[key][iso] !== undefined) {
                        exceptionResult = DATA.exceptions[key][iso];
                        break;
                    }
                }
            }
            if (exceptionResult !== undefined) return exceptionResult;
        }
    }

    const pCode = patternCode || (DATA.viewer ? DATA.viewer.pattern : null);
    if (!DATA.patterns || !DATA.patterns[pCode]) { 
        const d = date.getDay();
        return d !== 0 && d !== 6; 
    } 
    
    const d = new Date(date);
    const utcDate = Date.UTC(d.getFullYear(), d.getMonth(), d.getDate(), 12, 0, 0, 0);
    const utcEpoch = Date.UTC(2024, 11, 29, 12, 0, 0, 0);
    const diffMs = utcDate - utcEpoch;
    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
    const cycleLength = DATA.patterns[pCode].cycleLength;
    const weekIndex = Math.floor(diffDays / 7) % cycleLength;
    const normalizedIndex = weekIndex >= 0 ? weekIndex : weekIndex + cycleLength;
    const weekDays = DATA.patterns[pCode].weeks[normalizedIndex];
    return weekDays && weekDays.includes(d.toLocaleDateString('en-GB', {weekday: 'short'}));
}

    function getUsersByDept(selectedArray) {
        const stockComplianceTeams = ["QC AM", "QC PM", "PI AM", "PI PM 1", "PI PM 2", "PI NT", "Stop & Check AM", "Stop & Check PM", "Stop & Check NT", "NLL AM", "Compliance Analyst", "QC/PI", "Continuous Improvement"];
        
        // NEW SPLIT SHUNTERS
        const transDrivers = [
            "Transport AM", "Transport PM", "Transport NT", 
            "Transport Shunter AM1", "Transport Shunter AM2", 
            "Transport Shunter PM1", "Transport Shunter PM2", "Transport Shunter PM3", 
            "Transport Shunter NT1", "Transport Shunter NT2"
        ];
        
        const transAdmin = ["Transport Trainer AM", "Transport Trainer PM", "Transport Admin Group 1", "Transport Admin Group 2", "Transport Admin Group 3", "Transport Admin Group 4", "Transport Line Manager AM", "Transport Line Manager PM", "Transport Line Manager NT"];
        
        // NEW SORT ORDER
        const transportPriority = { 
            "Transport Shunter AM1": 1, "Transport Shunter AM2": 2, "Transport AM": 3, 
            "Transport Shunter PM1": 4, "Transport Shunter PM2": 5, "Transport Shunter PM3": 6, "Transport PM": 7, 
            "Transport Shunter NT1": 8, "Transport Shunter NT2": 9, "Transport NT": 10 
        };

        // 1. FILTERING
        let filteredUsers = DATA.users.filter(u => {
            if (selectedArray.includes('All')) return true;
            
            // Check if user belongs to ANY of the selected groups/depts
            return selectedArray.some(sel => {
                if (sel === 'Stock Compliance') return stockComplianceTeams.includes(u.dept); 
                if (sel === 'Group_AM') return u.dept && u.dept.endsWith(" AM"); 
                if (sel === 'Group_PM') return u.dept && u.dept.endsWith(" PM"); 
                if (sel === 'Group_NT') return u.dept && u.dept.endsWith(" NT"); 
                if (sel === 'Group_OpsMgr') return u.dept && u.dept.toLowerCase().includes("ops manager");
                if (sel === 'Transport_Drivers') return transDrivers.includes(u.dept); 
                if (sel === 'Transport_Admin') return transAdmin.includes(u.dept);
                return u.dept === sel; 
            });
        });

        // 2. SORTING
        return filteredUsers.sort((a, b) => {
            const priorityA = transportPriority[a.dept] || 999;
            const priorityB = transportPriority[b.dept] || 999;
            if (priorityA !== priorityB) return priorityA - priorityB;
            if (a.dept !== b.dept) return a.dept.localeCompare(b.dept);
            return a.name.localeCompare(b.name);
        });
    }
